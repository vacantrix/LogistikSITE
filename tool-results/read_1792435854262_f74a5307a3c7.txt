    45→const THIN_SCROLL = '[&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-slate-300'

    47→type ModeFilter = 'ALL' | RequestMode | 'MY_BIDS'

    49→const MODE_FILTERS: { value: ModeFilter; label: string }[] = [
    50→  { value: 'ALL', label: 'Все статусы' },

    52→  { value: 'CLOSED', label: 'Закрытые ставки' },

    55→/** Только для роли перевозчика: заявки, куда он подавал ставку */

    58→/** Ставка в таблице: текущая лучшая и стартовая */
    59→function RateCell({ request }: { request: RequestDTO }) {
    60→  if (request.mode === 'CLOSED') {
    61→    return (
    62→      <div>
    63→        <div className="flex items-center gap-1 font-semibold text-slate-900">
    64→          <Lock className="h-3.5 w-3.5 text-slate-400" aria-hidden />
    65→          Скрыта
    66→        </div>
    67→        <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    68→      </div>
    69→    )
    70→  }
    71→  if (request.bestRate == null) {
    72→    return (
    73→      <div>
    74→        <div className="text-sm font-medium text-slate-500">Ставок нет</div>
    75→        <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    76→      </div>
    77→    )
    78→  }
    79→  return (
    80→    <div>
    81→      <div className="font-semibold text-slate-900">{money(request.bestRate)}</div>
    82→      <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    83→    </div>
    84→  )
    85→}
    86→
    87→/** Обратный отсчёт до окончания приёма ставок (без hydration mismatch) */
    88→function DeadlineCell({ request, now }: { request: RequestDTO; now: number | null }) {
    89→  if (!request.deadline) return <span className="text-slate-400">—</span>
    90→  if (now === null) return <span className="text-slate-400">—</span>
    91→  const tl = timeLeft(request.deadline)
    92→  if (!tl) return <span className="text-slate-400">—</span>
    93→  return (
    94→    <span className={`tabular-nums ${tl.urgent ? 'font-semibold text-red-600' : 'text-slate-700'}`}>
    95→      {tl.text}
    96→    </span>
    97→  )
    98→}

   100→export default function BoardTab({ activeCarrierId, role, onNavigate }: TabProps) {
   101→  const { toast } = useToast()
   102→  const [requests, setRequests] = useState<RequestDTO[] | null>(null)
   103→  const [loading, setLoading] = useState(true)
   104→  const [refreshing, setRefreshing] = useState(false)
   105→  const [error, setError] = useState<string | null>(null)
   106→  const [q, setQ] = useState('')
   107→  const [mode, setMode] = useState<ModeFilter>('ALL')
   108→  const [now, setNow] = useState<number | null>(null)
   109→  const isLogist = role === 'logist'
   110→
   111→  /** Пункты фильтра статуса: у перевозчика добавляется «Мои отклики» */
   112→  const statusFilters = useMemo(
   113→    () => (isLogist ? MODE_FILTERS : [...MODE_FILTERS, MY_BIDS_FILTER]),
   114→    [isLogist],
   115→  )
   116→
   117→  // если роль сменилась на логиста — сбрасываем «Мои отклики»
   118→  useEffect(() => {
   119→    if (isLogist && mode === 'MY_BIDS') setMode('ALL')
   120→  }, [isLogist, mode])
   121→
   122→  // диалоги: держим только id, свежие данные берём из списка
   123→  const [detailId, setDetailId] = useState<string | null>(null)
   124→  const [bidId, setBidId] = useState<string | null>(null)
   125→  const [bidAmount, setBidAmount] = useState('')
   126→  const [busyId, setBusyId] = useState<string | null>(null)
   127→
   128→  // создание заявки — только логист
   129→  const [createOpen, setCreateOpen] = useState(false)
   130→  const [creating, setCreating] = useState(false)
   131→  const [form, setForm] = useState({
   132→    fromPoint: '', toPoint: '', cargo: '', tonnage: '20', truckType: 'Тент',
   133→    mode: 'AUCTION' as RequestMode, startRate: '', loadDate: '', loadWindow: '08:00–12:00', deadlineHours: '4',
   134→  })
   135→
   136→  const load = useCallback(async (silent = false) => {
   137→    if (!silent) setRefreshing(true)
   138→    try {
   139→      const d = await api.listRequests()
   140→      setRequests(d.requests)
   141→      setError(null)
   142→    } catch (e) {
   143→      setError(e instanceof Error ? e.message : 'Неизвестная ошибка')
   144→    } finally {
   145→      setLoading(false)
   146→      if (!silent) setRefreshing(false)
   147→    }
   148→  }, [])
   149→
   150→  // первичная загрузка + автообновление каждые 15 секунд
   151→  useEffect(() => {
   152→    void load()
   153→    const t = setInterval(() => void load(true), 15_000)
   154→    return () => clearInterval(t)
   155→  }, [load])
   156→
   157→  // таймер для обратного отсчёта — только после монтирования
   158→  useEffect(() => {
   159→    setNow(Date.now())
   160→    const t = setInterval(() => setNow(Date.now()), 1000)
   161→    return () => clearInterval(t)
   162→  }, [])
   163→
   164→  const applyRequest = useCallback((updated: RequestDTO) => {
   165→    setRequests((prev) => {
   166→      const list = prev ?? []
   167→      const i = list.findIndex((r) => r.id === updated.id)
   168→      if (i === -1) return [updated, ...list]
   169→      const next = [...list]
   170→      next[i] = updated
   171→      return next
   172→    })
   173→  }, [])
   174→
   175→  /** Лучшая (минимальная) ставка активного перевозчика по заявке — для подсветки и фильтра */
   176→  const myBidOf = useCallback(
   177→    (r: RequestDTO): BidDTO | null => {
   178→      if (!activeCarrierId) return null
   179→      const mine = r.bids.filter((b) => b.carrierId === activeCarrierId)
   180→      if (mine.length === 0) return null
   181→      return mine.reduce((best, b) => (b.amount < best.amount ? b : best))
   182→    },
   183→    [activeCarrierId],
   184→  )
   185→
   186→  const filtered = useMemo(() => {
   187→    const list = requests ?? []
   188→    const needle = q.trim().toLowerCase()
   189→    return list.filter((r) => {
   190→      if (mode === 'MY_BIDS') {
   191→        if (!myBidOf(r)) return false
   192→      } else if (mode !== 'ALL' && r.mode !== mode) return false
   193→      if (!needle) return true
   194→      return [r.number, r.cargo, r.fromPoint, r.toPoint].some((v) => v.toLowerCase().includes(needle))
   195→    })
   196→  }, [requests, q, mode, myBidOf])
   197→
   198→  /** Выгрузка текущей выборки в Excel: то, что видно в таблице с учётом фильтров и поиска */
   199→  const exportExcel = () => {
   200→    const withMine = !isLogist
   201→    const header = [
   202→      'Заявка', 'Маршрут', 'Груз', 'Тоннаж, т', 'Тип ТС', 'Статус',
   203→      'Ставка', 'Стартовая, ₽', 'Погрузка', 'Окно', 'Приём ставок до',
   204→      ...(withMine ? ['Мой отклик'] : []),
   205→    ]
   206→    const rows = filtered.map((r) => {
   207→      const my = myBidOf(r)
   208→      const statusLabel = r.status === 'PUBLISHED'
   209→        ? (r.mode === 'AUCTION' ? 'Торги' : 'Закрытые ставки')
   210→        : REQUEST_STATUS_META[r.status].label
   211→      const rate = r.mode === 'CLOSED' ? 'Скрыта' : (r.bestRate != null ? r.bestRate : '—')
   212→      return [
   213→        r.number,
   214→        `${r.fromPoint} → ${r.toPoint}`,
   215→        r.cargo,
   216→        r.tonnage,
   217→        r.truckType,
   218→        statusLabel,
   219→        rate,
   220→        r.startRate,
   221→        r.loadDate,
   222→        r.loadWindow,
   223→        r.deadline ? fmtDateTime(r.deadline) : '—',
   224→        ...(withMine ? [my ? `Ваша ставка: ${my.amount} ₽` : '—'] : []),
   225→      ]
   226→    })
   227→    const cols = [{ wch: 10 }, { wch: 28 }, { wch: 22 }, { wch: 10 }, { wch: 9 }, { wch: 18 }, { wch: 13 }, { wch: 12 }, { wch: 12 }, { wch: 12 }, { wch: 17 }]
   228→    if (withMine) cols.push({ wch: 19 })
   229→    const ws = XLSX.utils.aoa_to_sheet([header, ...rows])
   230→    ws['!cols'] = cols
   231→    const wb = XLSX.utils.book_new()
   232→    XLSX.utils.book_append_sheet(wb, ws, 'Площадка')
   233→    XLSX.writeFile(wb, 'ploschadka-zayavok.xlsx')
   234→    toast({ title: 'Выгрузка готова', description: `В файле ${rows.length} заявок — текущая выборка с учётом фильтров и поиска` })
   235→  }
   236→
   237→  const openCount = (requests ?? []).filter((r) => r.status === 'PUBLISHED').length
   238→
   239→  const detailReq = (requests ?? []).find((r) => r.id === detailId) ?? null
   240→  const bidReq = (requests ?? []).find((r) => r.id === bidId) ?? null
   241→
   242→  const closeBid = () => {
   243→    setBidId(null)
   244→    setBidAmount('')
   245→  }
   246→
   247→  const doBid = async () => {
   248→    if (!bidReq || !activeCarrierId) return
   249→    const amount = Math.round(Number(bidAmount.replace(/\s/g, '').replace(',', '.')))
   250→    if (!amount || amount <= 0) {
   251→      toast({ title: 'Ошибка', description: 'Укажите ставку в рублях', variant: 'destructive' })
   252→      return
   253→    }
   254→    if (bidReq.mode === 'AUCTION') {
   255→      const floor = bidReq.bestRate ?? bidReq.startRate
   256→      if (amount >= floor) {
   257→        toast({ title: 'Ошибка', description: `Ставка должна быть ниже текущей лучшей (${money(floor)})`, variant: 'destructive' })
   258→        return
   259→      }
   260→    }
   261→    setBusyId(bidReq.id)
   262→    try {
   263→      const { request } = await api.action(bidReq.id, 'bid', { carrierId: activeCarrierId, amount })
   264→      applyRequest(request)
   265→      toast({ title: 'Ставка принята', description: `${request.number} — ${money(amount)}${request.mode === 'CLOSED' ? ' · закрытая ставка' : ''}` })
   266→      closeBid()
   267→    } catch (e) {
   268→      toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Не удалось подать ставку', variant: 'destructive' })
   269→    } finally {
   270→      setBusyId(null)
   271→    }
   272→  }
   273→
   274→  const sortedBids = (bids: BidDTO[]) => [...bids].sort((a, b) => a.amount - b.amount)
   275→  const bestBid = (bids: BidDTO[]) => (bids.length > 0 ? Math.min(...bids.map((b) => b.amount)) : null)
   276→
   277→  // ——— публикация заявки (только логист) ———
   278→  const doCreate = async () => {
   279→    if (!form.fromPoint.trim() || !form.toPoint.trim() || !form.cargo.trim() || !form.startRate.trim()) {
   280→      toast({ title: 'Ошибка', description: 'Заполните направление, груз и стартовую ставку', variant: 'destructive' })
   281→      return
   282→    }
   283→    setCreating(true)
   284→    try {
   285→      const { request } = await api.createRequest({
   286→        fromPoint: form.fromPoint.trim(),
   287→        toPoint: form.toPoint.trim(),
   288→        cargo: form.cargo.trim(),
   289→        tonnage: Number(form.tonnage) || 20,
   290→        truckType: form.truckType,
   291→        mode: form.mode,
   292→        startRate: Number(form.startRate.replace(/\s/g, '')),
   293→        loadDate: form.loadDate.trim() || '—',
   294→        loadWindow: form.loadWindow.trim() || '08:00–12:00',
   295→        deadline: new Date(Date.now() + Number(form.deadlineHours || 4) * 3600_000).toISOString(),
   296→      })
   297→      setRequests((prev) => [request, ...(prev ?? [])])
   298→      setCreateOpen(false)
   299→      setForm((f) => ({ ...f, fromPoint: '', toPoint: '', cargo: '', startRate: '' }))
   300→      toast({ title: 'Заявка опубликована', description: `${request.number}: ${request.fromPoint} → ${request.toPoint}` })
   301→    } catch (e) {
   302→      toast({ title: 'Не удалось опубликовать заявку', description: e instanceof Error ? e.message : 'Неизвестная ошибка', variant: 'destructive' })
   303→    } finally {
   304→      setCreating(false)
   305→    }
   306→  }
   307→
   308→  return (
   309→    <section aria-label="Площадка заявок">
   310→      <SectionHeader title="Площадка заявок">
   311→        <div className="flex items-center gap-3">
   312→          {isLogist && (
   313→            <Button onClick={() => setCreateOpen(true)} className="gap-1.5" aria-label="Создать и опубликовать заявку">
   314→              <Plus className="h-4 w-4" aria-hidden />
   315→              Создать заявку
   316→            </Button>
   317→          )}
   318→          <div className="text-sm font-medium text-slate-700" aria-live="polite">
   319→            Открыто <span className="text-lg font-bold text-blue-700">{openCount}</span> заявок
   320→          </div>
   321→        </div>
   322→      </SectionHeader>
   323→
   324→      {!isLogist && !activeCarrierId && (
   325→        <div className="mb-4 rounded-xl border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-900" role="status">
   326→          Выберите перевозчика в шапке, чтобы действовать от его имени
   327→        </div>
   328→      )}
   329→
   330→      {loading && <LoadingBlock label="Загружаем заявки…" />}
   331→      {!loading && error && <ErrorBlock message={error} onRetry={() => void load()} />}
   332→
   333→      {!loading && !error && requests && (
   334→        <div className="space-y-4">
   335→          {/* Фильтры */}
   336→          <Card className="border-slate-200 shadow-sm">
   337→            <CardContent className="flex flex-wrap items-center gap-3 p-4">
   338→              <div className="relative min-w-[220px] flex-1">
   339→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" aria-hidden />
   340→                <Input
   341→                  value={q}
   342→                  onChange={(e) => setQ(e.target.value)}
   343→                  placeholder="Поиск: номер, груз, маршрут…"
   344→                  className="pl-9"
   345→                  aria-label="Поиск по номеру, грузу или маршруту"
   346→                />
   347→              </div>
   348→              <Select value={mode} onValueChange={(v) => setMode(v as ModeFilter)}>
   349→                <SelectTrigger className="w-[200px]" aria-label="Фильтр по статусу заявки">
   350→                  <SelectValue placeholder="Все статусы" />
   351→                </SelectTrigger>
   352→                <SelectContent>
   353→                  {statusFilters.map((m) => (
   354→                    <SelectItem key={m.value} value={m.value}>{m.label}</SelectItem>
   355→                  ))}
   356→                </SelectContent>
   357→              </Select>
   358→              <Button variant="outline" onClick={exportExcel} disabled={filtered.length === 0} className="gap-2" aria-label="Выгрузить выборку в Excel">
   359→                <FileSpreadsheet className="h-4 w-4 text-emerald-600" aria-hidden />
   360→                В Excel
   361→              </Button>
   362→              <Button variant="outline" onClick={() => void load()} disabled={refreshing} className="gap-2" aria-label="Обновить список заявок">
   363→                <RefreshCw className={`h-4 w-4 ${refreshing ? 'animate-spin' : ''}`} aria-hidden />
   364→                Обновить
   365→              </Button>
   366→              <span className="ml-auto hidden text-xs text-muted-foreground sm:block">автообновление каждые 15 секунд</span>
   367→            </CardContent>
   368→          </Card>
   369→
   370→          {/* Таблица заявок */}
   371→          <Card className="border-slate-200 shadow-sm">
   372→            <CardContent className="p-4 sm:p-6">
   373→              {filtered.length === 0 ? (
   374→                <EmptyState
   375→                  title={mode === 'MY_BIDS' ? 'Откликов пока нет' : 'Заявок не найдено'}
   376→                  hint={mode === 'MY_BIDS'
   377→                    ? 'Заявки, куда перевозчик подавал ставку, появятся в этом списке'
   378→                    : 'Измените параметры поиска или фильтр статуса'}
   379→                />
   380→              ) : (
   381→                <div className="-mx-2 overflow-x-auto px-2">
   382→                  <table className="w-full min-w-[980px] border-separate border-spacing-0 text-sm">
   383→                    <thead>
   384→                      <tr className="text-left text-xs uppercase tracking-wide text-slate-500">
   385→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">№ заявки</th>
   386→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Маршрут</th>
   387→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Груз</th>
   388→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Погрузка</th>
   389→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Тип ТС</th>
   390→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Статус</th>
   391→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Ставка</th>
   392→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">До окончания</th>
   393→                        <th scope="col" className="border-b border-slate-200 pb-2.5 font-semibold">Действие</th>
   394→                      </tr>
   395→                    </thead>
   396→                    <tbody>
   397→                      {filtered.map((r) => {
   398→                        const my = myBidOf(r)
   399→                        return (
   400→                          <tr
   401→                            key={r.id}
   402→                            onClick={() => setDetailId(r.id)}
   403→                            className={`cursor-pointer align-middle transition-colors hover:bg-blue-50/40 ${my ? 'bg-sky-50/70' : ''}`}
   404→                          >
   405→                            <td className="border-b border-slate-100 py-3 pr-3">
   406→                              <div className="flex flex-col items-start gap-1">
   407→                                <span className="font-semibold text-blue-700">{r.number}</span>
   408→                                {my && (
   409→                                  <Badge variant="outline" className="whitespace-nowrap border-sky-200 bg-sky-100 text-[10px] text-sky-800">
   410→                                    Ваша ставка: {money(my.amount)}
   411→                                  </Badge>
   412→                                )}
   413→                              </div>
   414→                            </td>
   415→                          <td className="border-b border-slate-100 py-3 pr-3">
   416→                            <span className="flex items-center gap-1.5 font-medium text-slate-800">
   417→                              {r.fromPoint}
   418→                              <ArrowRight className="h-3.5 w-3.5 shrink-0 text-slate-400" aria-hidden />
   419→                              {r.toPoint}
   420→                            </span>
   421→                          </td>
   422→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">
   423→                            {r.cargo}
   424→                            <span className="ml-1 whitespace-nowrap text-muted-foreground">{r.tonnage} т</span>
   425→                          </td>
   426→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">
   427→                            <div>{r.loadDate}</div>
   428→                            <div className="text-xs text-muted-foreground">{r.loadWindow}</div>
