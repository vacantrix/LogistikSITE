    70→/** Цвет рамки слота по вердикту нейросети */
    71→function slotBorder(s: SlotState): string {
    72→  if (s.checking) return 'border-blue-300 bg-blue-50/30'
    73→  if (s.error) return 'border-red-300 bg-red-50/40'
    74→  if (!s.verdict) return 'border-slate-200'
    75→  switch (s.verdict.status) {
    76→    case 'PASSED': return 'border-emerald-300 bg-emerald-50/40'
    77→    case 'MANUAL': return 'border-blue-300 bg-blue-50/30'
    78→    case 'REUPLOAD': return 'border-amber-300 bg-amber-50/40'
    79→    case 'REJECTED': return 'border-red-300 bg-red-50/40'
    80→    default: return 'border-slate-200'
    81→  }
    82→}
    83→
    84→function confCls(c: number): string {
    85→  if (c >= 0.95) return 'text-emerald-600'
    86→  if (c >= 0.6) return 'text-blue-600'
    87→  return 'text-amber-600'
    88→}

    90→/** Панель вердикта нейросети под загруженным документом */
    91→function SlotVerdict({ s, type }: { s: SlotState; type: DocType }) {
    92→  if (s.checking) {
    93→    return (
    94→      <div className="mt-2 flex items-center gap-2 rounded-lg bg-blue-50 px-2.5 py-2 text-xs font-medium text-blue-800" role="status">
    95→        <Loader2 className="h-3.5 w-3.5 animate-spin" aria-hidden />
    96→        Нейросеть проверяет вид документа и данные…
    97→      </div>
    98→    )
    99→  }
   100→  if (s.error) {
   101→    return (
   102→      <div className="mt-2 flex items-start gap-2 rounded-lg bg-red-50 px-2.5 py-2 text-xs text-red-700">
   103→        <XCircle className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   104→        {s.error}
   105→      </div>
   106→    )
   107→  }
   108→  const v = s.verdict
   109→  if (!v) {
   110→    return <p className="mt-2 text-[11px] leading-snug text-slate-400">{SLOT_META[type].hint}. Нейросеть проверит документ сразу после загрузки.</p>
   111→  }
   112→
   113→  if (v.status === 'REJECTED') {
   114→    return (
   115→      <div className="mt-2 space-y-1.5 rounded-lg border border-red-200 bg-red-50 px-2.5 py-2">
   116→        <div className="flex items-center gap-1.5 text-xs font-semibold text-red-800">
   117→          <XCircle className="h-3.5 w-3.5" aria-hidden />
   118→          Отказ нейросети — замените файл
   119→        </div>
   120→        <p className="text-[11px] leading-snug text-red-700">{v.message}</p>
   121→        <p className="text-[11px] leading-snug text-red-600">Заявка не будет отправлена, пока документ не заменён.</p>
   122→      </div>
   123→    )
   124→  }
   125→  if (v.status === 'REUPLOAD') {
   126→    return (
   127→      <div className="mt-2 space-y-1 rounded-lg border border-amber-200 bg-amber-50 px-2.5 py-2">
   128→        <div className="flex items-center gap-1.5 text-xs font-semibold text-amber-800">
   129→          <AlertTriangle className="h-3.5 w-3.5" aria-hidden />
   130→          <DocStatusBadge status={v.status} />
   131→        </div>
   132→        <p className="text-[11px] leading-snug text-amber-700">{v.message}</p>
   133→      </div>
   134→    )
   135→  }
   136→  if (v.status === 'MANUAL') {
   137→    return (
   138→      <div className="mt-2 space-y-1 rounded-lg border border-blue-200 bg-blue-50 px-2.5 py-2">
   139→        <div className="flex items-center gap-1.5 text-xs font-semibold text-blue-800">
   140→          <Info className="h-3.5 w-3.5" aria-hidden />
   141→          <DocStatusBadge status={v.status} />
   142→        </div>
   143→        <p className="text-[11px] leading-snug text-blue-700">{v.message}</p>
   144→      </div>
   145→    )
   146→  }
   147→  // PASSED: показываем распознанные поля с уверенностью
   148→  const fieldRows = Object.entries(v.fields)
   149→    .filter(([, val]) => val)
   150→    .slice(0, 3)
   151→  return (
   152→    <div className="mt-2 space-y-1 rounded-lg border border-emerald-200 bg-emerald-50 px-2.5 py-2">
   153→      <div className="flex items-center gap-1.5 text-xs font-semibold text-emerald-800">
   154→        <CheckCircle2 className="h-3.5 w-3.5" aria-hidden />
   155→        Вид документа подтверждён · качество {v.quality}
   156→      </div>
   157→      {fieldRows.map(([label, val]) => (
   158→        <div key={label} className="flex items-baseline justify-between gap-2 text-[11px]">
   159→          <span className="text-slate-500">{label}:</span>
   160→          <span className="min-w-0 truncate font-medium text-slate-700">{val}</span>
   161→          <span className={`shrink-0 font-semibold ${confCls(v.confidence[label] ?? 0)}`}>
   162→            {v.confidence[label] != null ? `${Math.round((v.confidence[label] ?? 0) * 100)}%` : ''}
   163→          </span>
   164→        </div>
   165→      ))}
   166→    </div>
   167→  )
   168→}
