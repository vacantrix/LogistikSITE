    76→/** Текст-подсказка для агента по статусу заявки */
    77→function agentLine(r: RequestDTO): { text: string; cls: string } {
    78→  switch (r.status) {
    79→    case 'AWARDED':
    80→      return { text: 'Укажите водителя и ТС, загрузите документы', cls: 'border-amber-200 bg-amber-50 text-amber-900' }
    81→    case 'DOCS_REVIEW': {
    82→      const bad = r.documents.some((d) => d.status === 'REUPLOAD' || d.status === 'REJECTED')
    83→      return bad
    84→        ? { text: 'Требует исправления: причина по каждому документу, догрузка — в ту же заявку', cls: 'border-amber-300 bg-amber-50 text-amber-900' }
    85→        : { text: 'Нейросеть проверяет документы', cls: 'border-blue-200 bg-blue-50 text-blue-900' }
    86→    }
    87→    case 'AT_LOGIST':
    88→      return { text: 'Комплект у логиста — идёт сверка фото с полями', cls: 'border-blue-200 bg-blue-50 text-blue-900' }
    89→    case 'CONFIRMED':
    90→      return { text: 'Принята — данные переданы в 1С', cls: 'border-emerald-200 bg-emerald-50 text-emerald-900' }
    91→    default:
    92→      return { text: 'Рейс закрыт', cls: 'border-slate-300 bg-slate-100 text-slate-600' }
    93→  }
    94→}

   111→/** Читает файл и сжимает до dataUri JPEG (max 1280px, качество 0.8) */
   112→async function fileToCompressedDataUri(file: File): Promise<string> {
   113→  const dataUrl = await new Promise<string>((resolve, reject) => {
   114→    const reader = new FileReader()
   115→    reader.onload = () => resolve(String(reader.result))
   116→    reader.onerror = () => reject(new Error('Не удалось прочитать файл'))
   117→    reader.readAsDataURL(file)
   118→  })
   119→  try {
   120→    const img = document.createElement('img')
   121→    img.decoding = 'async'
   122→    await new Promise<void>((resolve, reject) => {
   123→      img.onload = () => resolve()
   124→      img.onerror = () => reject(new Error('image'))
   125→      img.src = dataUrl
   126→    })
   127→    const max = 1280
   128→    const scale = Math.min(1, max / Math.max(img.width, img.height, 1))
   129→    const canvas = document.createElement('canvas')
   130→    canvas.width = Math.max(1, Math.round(img.width * scale))
   131→    canvas.height = Math.max(1, Math.round(img.height * scale))
   132→    const ctx = canvas.getContext('2d')
   133→    if (!ctx) return dataUrl
   134→    ctx.drawImage(img, 0, 0, canvas.width, canvas.height)
   135→    return canvas.toDataURL('image/jpeg', 0.8)
   136→  } catch {
   137→    return dataUrl
   138→  }
   139→}

   141→/** Демо-образец документа: SVG → dataUri */
   142→const sampleSvg = (title: string, lines: string[], accent: string) => {
   143→  const rows = lines
   144→    .map((l, i) => `<text x="40" y="${150 + i * 44}" font-family="Arial" font-size="24" fill="#1f2937">${l}</text>`)
   145→    .join('')
   146→  const svg = `<svg xmlns="http://www.w3.org/2000/svg" width="640" height="420"><rect width="640" height="420" rx="14" fill="#f8fafc" stroke="#cbd5e1"/><rect width="640" height="64" rx="14" fill="${accent}"/><text x="40" y="42" font-family="Arial" font-size="24" font-weight="bold" fill="#fff">${title}</text>${rows}</svg>`
   147→  return `data:image/svg+xml;base64,${btoa(unescape(encodeURIComponent(svg)))}`
   148→}
