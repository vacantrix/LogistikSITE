   221→/** Русское склонение: 1 заявка / 2 заявки / 5 заявок */
   222→function zayavokWord(n: number): string {
   223→  const m10 = n % 10
   224→  const m100 = n % 100
   225→  if (m10 === 1 && m100 !== 11) return 'заявка'
   226→  if (m10 >= 2 && m10 <= 4 && (m100 < 12 || m100 > 14)) return 'заявки'
   227→  return 'заявок'
   228→}
