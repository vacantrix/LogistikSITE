    17→import {

    43→const THIN_SCROLL = '[&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-slate-300'

    45→type ModeFilter = 'ALL' | RequestMode

    47→const MODE_FILTERS: { value: ModeFilter; label: string }[] = [
    48→  { value: 'ALL', label: 'Все статусы' },

    53→/** Ставка в таблице: текущая лучшая и стартовая */
    54→function RateCell({ request }: { request: RequestDTO }) {
    55→  if (request.mode === 'CLOSED') {
    56→    return (
    57→      <div>
    58→        <div className="flex items-center gap-1 font-semibold text-slate-900">
    59→          <Lock className="h-3.5 w-3.5 text-slate-400" aria-hidden />
    60→          Скрыта
    61→        </div>
    62→        <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    63→      </div>
    64→    )
    65→  }
    66→  if (request.bestRate == null) {
    67→    return (
    68→      <div>
    69→        <div className="text-sm font-medium text-slate-500">Ставок нет</div>
    70→        <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    71→      </div>
    72→    )
    73→  }
    74→  return (
    75→    <div>
    76→      <div className="font-semibold text-slate-900">{money(request.bestRate)}</div>
    77→      <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    78→    </div>
    79→  )
    80→}
    81→
    82→/** Обратный отсчёт до окончания приёма ставок (без hydration mismatch) */
    83→function DeadlineCell({ request, now }: { request: RequestDTO; now: number | null }) {
    84→  if (!request.deadline) return <span className="text-slate-400">—</span>
    85→  if (now === null) return <span className="text-slate-400">—</span>
    86→  const tl = timeLeft(request.deadline)
    87→  if (!tl) return <span className="text-slate-400">—</span>
    88→  return (
    89→    <span className={`tabular-nums ${tl.urgent ? 'font-semibold text-red-600' : 'text-slate-700'}`}>
    90→      {tl.text}
    91→    </span>
    92→  )
    93→}

    95→export default function BoardTab({ activeCarrierId, role, onNavigate }: TabProps) {
    96→  const { toast } = useToast()
    97→  const [requests, setRequests] = useState<RequestDTO[] | null>(null)
    98→  const [loading, setLoading] = useState(true)
    99→  const [refreshing, setRefreshing] = useState(false)
   100→  const [error, setError] = useState<string | null>(null)
   101→  const [q, setQ] = useState('')
   102→  const [mode, setMode] = useState<ModeFilter>('ALL')
   103→  const [now, setNow] = useState<number | null>(null)
   104→  const isLogist = role === 'logist'
   105→
   106→  // диалоги: держим только id, свежие данные берём из списка
   107→  const [detailId, setDetailId] = useState<string | null>(null)
   108→  const [bidId, setBidId] = useState<string | null>(null)
   109→  const [bidAmount, setBidAmount] = useState('')
   110→  const [busyId, setBusyId] = useState<string | null>(null)
   111→
   112→  // создание заявки — только логист
   113→  const [createOpen, setCreateOpen] = useState(false)
   114→  const [creating, setCreating] = useState(false)
   115→  const [form, setForm] = useState({
   116→    fromPoint: '', toPoint: '', cargo: '', tonnage: '20', truckType: 'Тент',
   117→    mode: 'AUCTION' as RequestMode, startRate: '', loadDate: '', loadWindow: '08:00–12:00', deadlineHours: '4',
   118→  })
   119→
   120→  const load = useCallback(async (silent = false) => {
   121→    if (!silent) setRefreshing(true)
   122→    try {
   123→      const d = await api.listRequests()
   124→      setRequests(d.requests)
   125→      setError(null)
   126→    } catch (e) {
   127→      setError(e instanceof Error ? e.message : 'Неизвестная ошибка')
   128→    } finally {
   129→      setLoading(false)
   130→      if (!silent) setRefreshing(false)
   131→    }
   132→  }, [])
   133→
   134→  // первичная загрузка + автообновление каждые 15 секунд
   135→  useEffect(() => {
   136→    void load()
   137→    const t = setInterval(() => void load(true), 15_000)
   138→    return () => clearInterval(t)
   139→  }, [load])
   140→
   141→  // таймер для обратного отсчёта — только после монтирования
   142→  useEffect(() => {
   143→    setNow(Date.now())
   144→    const t = setInterval(() => setNow(Date.now()), 1000)
   145→    return () => clearInterval(t)
   146→  }, [])
   147→
   148→  const applyRequest = useCallback((updated: RequestDTO) => {
   149→    setRequests((prev) => {
   150→      const list = prev ?? []
   151→      const i = list.findIndex((r) => r.id === updated.id)
   152→      if (i === -1) return [updated, ...list]
   153→      const next = [...list]
   154→      next[i] = updated
   155→      return next
   156→    })
   157→  }, [])
   158→
   159→  const filtered = useMemo(() => {
   160→    const list = requests ?? []
   161→    const needle = q.trim().toLowerCase()
   162→    return list.filter((r) => {
   163→      if (mode !== 'ALL' && r.mode !== mode) return false
   164→      if (!needle) return true
   165→      return [r.number, r.cargo, r.fromPoint, r.toPoint].some((v) => v.toLowerCase().includes(needle))
   166→    })
   167→  }, [requests, q, mode])
   168→
   169→  const openCount = (requests ?? []).filter((r) => r.status === 'PUBLISHED').length
   170→
   171→  const detailReq = (requests ?? []).find((r) => r.id === detailId) ?? null
   172→  const bidReq = (requests ?? []).find((r) => r.id === bidId) ?? null
   173→
   174→  const closeBid = () => {
   175→    setBidId(null)
   176→    setBidAmount('')
   177→  }
   178→
   179→  const doBid = async () => {
   180→    if (!bidReq || !activeCarrierId) return
   181→    const amount = Math.round(Number(bidAmount.replace(/\s/g, '').replace(',', '.')))
   182→    if (!amount || amount <= 0) {
   183→      toast({ title: 'Ошибка', description: 'Укажите ставку в рублях', variant: 'destructive' })
   184→      return
   185→    }
   186→    if (bidReq.mode === 'AUCTION') {
   187→      const floor = bidReq.bestRate ?? bidReq.startRate
   188→      if (amount >= floor) {
   189→        toast({ title: 'Ошибка', description: `Ставка должна быть ниже текущей лучшей (${money(floor)})`, variant: 'destructive' })
   190→        return
   191→      }
   192→    }
   193→    setBusyId(bidReq.id)
   194→    try {
   195→      const { request } = await api.action(bidReq.id, 'bid', { carrierId: activeCarrierId, amount })
   196→      applyRequest(request)
   197→      toast({ title: 'Ставка принята', description: `${request.number} — ${money(amount)}${request.mode === 'CLOSED' ? ' · закрытая ставка' : ''}` })
   198→      closeBid()
   199→    } catch (e) {
   200→      toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Не удалось подать ставку', variant: 'destructive' })
   201→    } finally {
   202→      setBusyId(null)
   203→    }
   204→  }
   205→
   206→  const sortedBids = (bids: BidDTO[]) => [...bids].sort((a, b) => a.amount - b.amount)
   207→  const bestBid = (bids: BidDTO[]) => (bids.length > 0 ? Math.min(...bids.map((b) => b.amount)) : null)
   208→
   209→  // ——— публикация заявки (только логист) ———
   210→  const doCreate = async () => {
   211→    if (!form.fromPoint.trim() || !form.toPoint.trim() || !form.cargo.trim() || !form.startRate.trim()) {
   212→      toast({ title: 'Ошибка', description: 'Заполните направление, груз и стартовую ставку', variant: 'destructive' })
   213→      return
   214→    }
   215→    setCreating(true)
   216→    try {
   217→      const { request } = await api.createRequest({
   218→        fromPoint: form.fromPoint.trim(),
   219→        toPoint: form.toPoint.trim(),
   220→        cargo: form.cargo.trim(),
   221→        tonnage: Number(form.tonnage) || 20,
   222→        truckType: form.truckType,
   223→        mode: form.mode,
   224→        startRate: Number(form.startRate.replace(/\s/g, '')),
   225→        loadDate: form.loadDate.trim() || '—',
   226→        loadWindow: form.loadWindow.trim() || '08:00–12:00',
   227→        deadline: new Date(Date.now() + Number(form.deadlineHours || 4) * 3600_000).toISOString(),
   228→      })
   229→      setRequests((prev) => [request, ...(prev ?? [])])
   230→      setCreateOpen(false)
   231→      setForm((f) => ({ ...f, fromPoint: '', toPoint: '', cargo: '', startRate: '' }))
   232→      toast({ title: 'Заявка опубликована', description: `${request.number}: ${request.fromPoint} → ${request.toPoint}` })
   233→    } catch (e) {
   234→      toast({ title: 'Не удалось опубликовать заявку', description: e instanceof Error ? e.message : 'Неизвестная ошибка', variant: 'destructive' })
   235→    } finally {
   236→      setCreating(false)
   237→    }
   238→  }
   239→
   240→  return (
   241→    <section aria-label="Площадка заявок">
   242→      <SectionHeader title="Площадка заявок">
   243→        <div className="flex items-center gap-3">
   244→          {isLogist && (
   245→            <Button onClick={() => setCreateOpen(true)} className="gap-1.5" aria-label="Создать и опубликовать заявку">
   246→              <Plus className="h-4 w-4" aria-hidden />
   247→              Создать заявку
   248→            </Button>
   249→          )}
   250→          <div className="text-sm font-medium text-slate-700" aria-live="polite">
   251→            Открыто <span className="text-lg font-bold text-blue-700">{openCount}</span> заявок
   252→          </div>
   253→        </div>
   254→      </SectionHeader>
   255→
   256→      {!isLogist && !activeCarrierId && (
   257→        <div className="mb-4 rounded-xl border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-900" role="status">
   258→          Выберите перевозчика в шапке, чтобы действовать от его имени
   259→        </div>
   260→      )}
   261→
   262→      {loading && <LoadingBlock label="Загружаем заявки…" />}
   263→      {!loading && error && <ErrorBlock message={error} onRetry={() => void load()} />}
   264→
   265→      {!loading && !error && requests && (
   266→        <div className="space-y-4">
   267→          {/* Фильтры */}
   268→          <Card className="border-slate-200 shadow-sm">
   269→            <CardContent className="flex flex-wrap items-center gap-3 p-4">
   270→              <div className="relative min-w-[220px] flex-1">
   271→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" aria-hidden />
   272→                <Input
   273→                  value={q}
   274→                  onChange={(e) => setQ(e.target.value)}
   275→                  placeholder="Поиск: номер, груз, маршрут…"
   276→                  className="pl-9"
   277→                  aria-label="Поиск по номеру, грузу или маршруту"
   278→                />
   279→              </div>
   280→              <Select value={mode} onValueChange={(v) => setMode(v as ModeFilter)}>
   281→                <SelectTrigger className="w-[200px]" aria-label="Фильтр по статусу заявки">
   282→                  <SelectValue placeholder="Все статусы" />
   283→                </SelectTrigger>
   284→                <SelectContent>
   285→                  {MODE_FILTERS.map((m) => (
   286→                    <SelectItem key={m.value} value={m.value}>{m.label}</SelectItem>
   287→                  ))}
   288→                </SelectContent>
   289→              </Select>
   290→              <Button variant="outline" onClick={() => void load()} disabled={refreshing} className="gap-2" aria-label="Обновить список заявок">
   291→                <RefreshCw className={`h-4 w-4 ${refreshing ? 'animate-spin' : ''}`} aria-hidden />
   292→                Обновить
   293→              </Button>
   294→              <span className="ml-auto hidden text-xs text-muted-foreground sm:block">автообновление каждые 15 секунд</span>
   295→            </CardContent>
   296→          </Card>
   297→
   298→          {/* Таблица заявок */}
   299→          <Card className="border-slate-200 shadow-sm">
   300→            <CardContent className="p-4 sm:p-6">
   301→              {filtered.length === 0 ? (
   302→                <EmptyState
   303→                  title="Заявок не найдено"
   304→                  hint="Измените параметры поиска или фильтр статуса"
   305→                />
   306→              ) : (
   307→                <div className="-mx-2 overflow-x-auto px-2">
   308→                  <table className="w-full min-w-[980px] border-separate border-spacing-0 text-sm">
   309→                    <thead>
   310→                      <tr className="text-left text-xs uppercase tracking-wide text-slate-500">
   311→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">№ заявки</th>
   312→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Маршрут</th>
   313→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Груз</th>
   314→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Погрузка</th>
   315→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Тип ТС</th>
   316→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Статус</th>
   317→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Ставка</th>
   318→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">До окончания</th>
   319→                        <th scope="col" className="border-b border-slate-200 pb-2.5 font-semibold">Действие</th>
   320→                      </tr>
   321→                    </thead>
   322→                    <tbody>
   323→                      {filtered.map((r) => (
   324→                        <tr
   325→                          key={r.id}
   326→                          onClick={() => setDetailId(r.id)}
   327→                          className="cursor-pointer align-middle transition-colors hover:bg-blue-50/40"
   328→                        >
   329→                          <td className="border-b border-slate-100 py-3 pr-3">
   330→                            <span className="font-semibold text-blue-700">{r.number}</span>
   331→                          </td>
   332→                          <td className="border-b border-slate-100 py-3 pr-3">
   333→                            <span className="flex items-center gap-1.5 font-medium text-slate-800">
   334→                              {r.fromPoint}
   335→                              <ArrowRight className="h-3.5 w-3.5 shrink-0 text-slate-400" aria-hidden />
   336→                              {r.toPoint}
   337→                            </span>
   338→                          </td>
   339→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">
   340→                            {r.cargo}
   341→                            <span className="ml-1 whitespace-nowrap text-muted-foreground">{r.tonnage} т</span>
   342→                          </td>
   343→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">
   344→                            <div>{r.loadDate}</div>
   345→                            <div className="text-xs text-muted-foreground">{r.loadWindow}</div>
   346→                          </td>
   347→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">{r.truckType}</td>
   348→                          <td className="border-b border-slate-100 py-3 pr-3"><ModeBadge mode={r.mode} /></td>
   349→                          <td className="border-b border-slate-100 py-3 pr-3"><RateCell request={r} /></td>
   350→                          <td className="border-b border-slate-100 py-3 pr-3"><DeadlineCell request={r} now={now} /></td>
   351→                          <td className="border-b border-slate-100 py-3" onClick={(e) => e.stopPropagation()}>
   352→                            {isLogist ? (
   353→                              // логист: не участвует в ставках — переходит в торги или смотрит статус
   354→                              <>
   355→                                {r.status === 'PUBLISHED' && r.mode === 'AUCTION' && (
   356→                                  <Button size="sm" variant="outline" className="gap-1.5" onClick={() => onNavigate('auctions')} aria-label={`Перейти к торгам по заявке ${r.number}`}>
   357→                                    <Gavel className="h-4 w-4" aria-hidden />
   358→                                    К торгам
   359→                                  </Button>
   360→                                )}
   361→                                {r.status === 'PUBLISHED' && r.mode !== 'AUCTION' && (
   362→                                  <span className="text-xs text-muted-foreground">ожидание перевозчика</span>
   363→                                )}
   364→                                {r.status !== 'PUBLISHED' && <StatusBadge status={r.status} />}
   365→                              </>
   366→                            ) : (
   367→                              <>
   368→                                {r.status === 'PUBLISHED' && r.mode === 'AUCTION' && (
   369→                                  <Button
   370→                                    size="sm"
   371→                                    disabled={!activeCarrierId || busyId !== null}
   372→                                    onClick={() => { setBidId(r.id); setBidAmount('') }}
   373→                                    aria-label={`Сделать ставку по заявке ${r.number}`}
   374→                                  >
   375→                                    <Gavel className="h-4 w-4" aria-hidden />
   376→                                    Сделать ставку
   377→                                  </Button>
   378→                                )}
   379→                                {r.status === 'PUBLISHED' && r.mode === 'CLOSED' && (
   380→                                  <Button
   381→                                    size="sm"
   382→                                    disabled={!activeCarrierId || busyId !== null}
   383→                                    onClick={() => { setBidId(r.id); setBidAmount('') }}
   384→                                    aria-label={`Подать закрытую ставку по заявке ${r.number}`}
   385→                                  >
   386→                                    <Lock className="h-4 w-4" aria-hidden />
   387→                                    Подать закрытую ставку
   388→                                  </Button>
   389→                                )}
   390→                                {r.status !== 'PUBLISHED' && <StatusBadge status={r.status} />}
   391→                              </>
   392→                            )}
   393→                          </td>
   394→                        </tr>
   395→                      ))}
   396→                    </tbody>
   397→                  </table>
   398→                </div>
   399→              )}
   400→            </CardContent>
   401→          </Card>
   402→        </div>
   403→      )}
   404→
   405→      {/* Диалог создания заявки — только логист */}
   406→      {isLogist && (
   407→        <Dialog open={createOpen} onOpenChange={setCreateOpen}>
   408→          <DialogContent className="max-h-[88vh] overflow-y-auto sm:max-w-lg">
   409→            <DialogHeader>
   410→              <DialogTitle>Новая заявка на перевозку</DialogTitle>
   411→              <DialogDescription>
   412→                Заявка будет опубликована на площадке с присвоением сквозного номера Z-XXXX.
   413→              </DialogDescription>
   414→            </DialogHeader>
   415→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   416→              <div className="space-y-1.5">
   417→                <Label htmlFor="cr-from">Пункт погрузки</Label>
   418→                <Input id="cr-from" placeholder="Абинск, АЭМЗ" value={form.fromPoint} onChange={(e) => setForm((f) => ({ ...f, fromPoint: e.target.value }))} />
   419→              </div>
   420→              <div className="space-y-1.5">
   421→                <Label htmlFor="cr-to">Пункт выгрузки</Label>
   422→                <Input id="cr-to" placeholder="Порт НМТП" value={form.toPoint} onChange={(e) => setForm((f) => ({ ...f, toPoint: e.target.value }))} />
   423→              </div>
   424→              <div className="space-y-1.5 sm:col-span-2">
   425→                <Label htmlFor="cr-cargo">Груз</Label>
   426→                <Input id="cr-cargo" placeholder="Оборудование" value={form.cargo} onChange={(e) => setForm((f) => ({ ...f, cargo: e.target.value }))} />
   427→              </div>
   428→              <div className="space-y-1.5">
   429→                <Label htmlFor="cr-tonnage">Тоннаж, т</Label>
   430→                <Input id="cr-tonnage" type="number" min={1} value={form.tonnage} onChange={(e) => setForm((f) => ({ ...f, tonnage: e.target.value }))} />
   431→              </div>
   432→              <div className="space-y-1.5">
   433→                <Label htmlFor="cr-truck">Тип ТС</Label>

   436→                  <SelectContent>
   437→                    {['Тент', 'Реф', 'Изотерм', 'Борт'].map((t) => <SelectItem key={t} value={t}>{t}</SelectItem>)}

   439→                </Select>
   440→              </div>
   441→              <div className="space-y-1.5">
   442→                <Label htmlFor="cr-mode">Статус</Label>

   452→                <Label htmlFor="cr-rate">Стартовая ставка, ₽</Label>

   454→              </div>
   455→              <div className="space-y-1.5">
   456→                <Label htmlFor="cr-date">Дата погрузки</Label>

   462→              </div>
   463→              <div className="space-y-1.5">
   464→                <Label htmlFor="cr-deadline">Приём ставок ещё</Label>
   465→                <Select value={form.deadlineHours} onValueChange={(v) => setForm((f) => ({ ...f, deadlineHours: v }))}>

   467→                  <SelectContent>
   468→                    {[['2', '2 часа'], ['4', '4 часа'], ['8', '8 часов'], ['24', '24 часа']].map(([v, l]) => (
   469→                      <SelectItem key={v} value={v}>{l}</SelectItem>
   470→                    ))}
   471→                  </SelectContent>
   472→                </Select>
   473→              </div>
   474→            </div>
   475→            <DialogFooter>
   476→              <Button variant="outline" onClick={() => setCreateOpen(false)}>Отмена</Button>
   477→              <Button onClick={() => void doCreate()} disabled={creating}>
   478→                {creating ? 'Публикуем…' : 'Опубликовать заявку'}
   479→              </Button>
   480→            </DialogFooter>
   481→          </DialogContent>
   482→        </Dialog>
   483→      )}
   484→
   485→      {/* Диалог ставки (торги / закрытые ставки) */}
   486→      <Dialog open={bidReq !== null} onOpenChange={(o) => { if (!o) closeBid() }}>
   487→        <DialogContent className="sm:max-w-md">
   488→          {bidReq && (
   489→            <>
   490→              <DialogHeader>
   491→                <DialogTitle>
   492→                  {bidReq.mode === 'CLOSED' ? 'Закрытая ставка' : 'Ставка'} · заявка {bidReq.number}

   513→                  <Input
   514→                    id="bid-amount"
   515→                    type="number"
   516→                    min={1}
   517→                    step={100}
   518→                    inputMode="numeric"
   519→                    placeholder="Например, 85000"
   520→                    value={bidAmount}
   521→                    onChange={(e) => setBidAmount(e.target.value)}
   522→                    onKeyDown={(e) => { if (e.key === 'Enter') void doBid() }}
   523→                  />
   524→                  <p className="text-xs text-muted-foreground">
   525→                    {bidReq.mode === 'CLOSED' ? 'Ставку увидит только логист при подведении итогов' : 'Должна быть ниже лучшей'}
   526→                  </p>
   527→                </div>
   528→              </div>
   529→              <DialogFooter>
   530→                <Button variant="outline" onClick={closeBid}>Отмена</Button>
   531→                <Button onClick={() => void doBid()} disabled={busyId !== null || !bidAmount.trim()}>
   532→                  {busyId === bidReq.id ? 'Отправляем…' : 'Подтвердить ставку'}
   533→                </Button>
   534→              </DialogFooter>
   535→            </>
   536→          )}
   537→        </DialogContent>
   538→      </Dialog>
   539→
   540→      {/* Детали заявки */}
   541→      <Dialog open={detailReq !== null} onOpenChange={(o) => { if (!o) setDetailId(null) }}>
   542→        <DialogContent className={`max-h-[90vh] overflow-y-auto sm:max-w-2xl ${THIN_SCROLL}`}>
   543→          {detailReq && (
   544→            <>
   545→              <DialogHeader>
   546→                <DialogTitle className="flex flex-wrap items-center gap-2">
   547→                  Заявка <span className="text-blue-700">{detailReq.number}</span>
   548→                  <StatusBadge status={detailReq.status} />
   549→                  <ModeBadge mode={detailReq.mode} />
   550→                </DialogTitle>
   551→                <DialogDescription className="flex items-center gap-1.5">
   552→                  {detailReq.fromPoint} <ArrowRight className="h-3.5 w-3.5" aria-hidden /> {detailReq.toPoint}
   553→                </DialogDescription>
   554→              </DialogHeader>
   555→
   556→              <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   557→                {[
   558→                  { label: 'Груз', value: `${detailReq.cargo}, ${detailReq.tonnage} т` },
   559→                  { label: 'Тип ТС', value: detailReq.truckType },
   560→                  { label: 'Погрузка', value: `${detailReq.loadDate}, ${detailReq.loadWindow}` },
   561→                  { label: 'Стартовая ставка', value: money(detailReq.startRate) },
   562→                  { label: 'Текущая лучшая', value: detailReq.bestRate != null ? money(detailReq.bestRate) : '—' },
   563→                  { label: detailReq.winnerName ? 'Перевозчик' : 'Окончание ставок', value: detailReq.winnerName ?? '—' },
   564→                ].map((f) => (
   565→                  <div key={f.label} className="rounded-lg border border-slate-200 bg-slate-50/60 p-2.5">
   566→                    <div className="text-[11px] uppercase tracking-wide text-muted-foreground">{f.label}</div>
   567→                    <div className="mt-0.5 text-sm font-semibold text-slate-900">{f.value}</div>
   568→                  </div>
   569→                ))}
   570→              </div>
   571→
   572→              <div className="rounded-lg border border-blue-200 bg-blue-50 p-3">
   573→                <div className="flex items-center gap-1.5 text-sm font-semibold text-blue-900">
   574→                  <ShieldCheck className="h-4 w-4" aria-hidden />
   575→                  Требования к документам
   576→                </div>
   577→                <p className="mt-1 text-sm text-blue-800">
   578→                  Пропуск в порт: паспорт, ВУ, СТС. Нейросеть сверит фото с данными водителя и автомобиля —
   579→                  комплект загружается в эту же заявку.
   580→                </p>
   581→              </div>
   582→
   583→              {detailReq.mode === 'AUCTION' && (
   584→                <div>
   585→                  <h4 className="mb-2 flex items-center gap-1.5 text-sm font-semibold text-slate-900">
   586→                    <Gavel className="h-4 w-4 text-blue-600" aria-hidden />
   587→                    Ставки ({detailReq.bids.length})
   588→                  </h4>
   589→                  {detailReq.bids.length === 0 ? (
   590→                    <p className="rounded-lg border border-dashed border-slate-300 px-3 py-4 text-center text-sm text-muted-foreground">
   591→                      Ставок пока нет — будьте первым.
   592→                    </p>
   593→                  ) : (
   594→                    <div className="overflow-x-auto rounded-lg border border-slate-200">
   595→                      <Table>
   596→                        <TableHeader>
   597→                          <TableRow>
   598→                            <TableHead>Перевозчик</TableHead>
   599→                            <TableHead>Ставка</TableHead>
   600→                            <TableHead>Время</TableHead>
   601→                          </TableRow>
   602→                        </TableHeader>
   603→                        <TableBody>
   604→                          {sortedBids(detailReq.bids).map((b) => {
   605→                            const isBest = bestBid(detailReq.bids) === b.amount
   606→                            return (
   607→                              <TableRow key={b.id} className={isBest ? 'bg-emerald-50' : undefined}>
   608→                                <TableCell className="font-medium">
   609→                                  <span className="flex items-center gap-2">
   610→                                    {b.carrierName}
   611→                                    {isBest && <Badge className="border-emerald-200 bg-emerald-100 text-[10px] text-emerald-800" variant="outline">лучшая</Badge>}
   612→                                  </span>
   613→                                </TableCell>
   614→                                <TableCell className={isBest ? 'font-semibold text-emerald-700' : 'text-slate-700'}>
   615→                                  {money(b.amount)}
   616→                                </TableCell>
   617→                                <TableCell className="text-muted-foreground">{fmtDateTime(b.createdAt)}</TableCell>
   618→                              </TableRow>
   619→                            )
   620→                          })}
   621→                        </TableBody>
   622→                      </Table>
   623→                    </div>
   624→                  )}
   625→                </div>
   626→              )}
   627→
   628→              {detailReq.mode === 'CLOSED' && (
   629→                <div className="flex items-center gap-2 rounded-lg border border-slate-200 bg-slate-50 p-3 text-sm text-slate-600">
   630→                  <Lock className="h-4 w-4 shrink-0" aria-hidden />
   631→                  Ставки скрыты до подведения итогов — участники не видят цены друг друга.
   632→                </div>
   633→              )}
   634→
   635→              {detailReq.status !== 'PUBLISHED' && detailReq.mode === 'AUCTION' && detailReq.bids.length > 0 && (
   636→                <p className="text-xs text-muted-foreground">
   637→                  Торги завершены — победитель согласован логистом.
   638→                </p>
   639→              )}
   640→            </>
   641→          )}
   642→        </DialogContent>
   643→      </Dialog>
