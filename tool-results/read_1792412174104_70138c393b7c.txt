   111→   111→   111→/** Читает файл и сжимает до dataUri JPEG (max 1280px, качество 0.8) */
   112→   112→   112→async function fileToCompressedDataUri(file: File): Promise<string> {
   113→   113→   113→  const dataUrl = await new Promise<string>((resolve, reject) => {
   114→   114→   114→    const reader = new FileReader()
   115→   115→   115→    reader.onload = () => resolve(String(reader.result))
   116→   116→   116→    reader.onerror = () => reject(new Error('Не удалось прочитать файл'))
   117→   117→   117→    reader.readAsDataURL(file)
   118→   118→   118→  })
   119→   119→   119→  try {
   120→   120→   120→    const img = document.createElement('img')
   121→   121→   121→    img.decoding = 'async'
   122→   122→   122→    await new Promise<void>((resolve, reject) => {
   123→   123→   123→      img.onload = () => resolve()
   124→   124→   124→      img.onerror = () => reject(new Error('image'))
   125→   125→   125→      img.src = dataUrl
   126→   126→   126→    })
   127→   127→   127→    const max = 1280
   128→   128→   128→    const scale = Math.min(1, max / Math.max(img.width, img.height, 1))
   129→   129→   129→    const canvas = document.createElement('canvas')
   130→   130→   130→    canvas.width = Math.max(1, Math.round(img.width * scale))
   131→   131→   131→    canvas.height = Math.max(1, Math.round(img.height * scale))
   132→   132→   132→    const ctx = canvas.getContext('2d')
   133→   133→   133→    if (!ctx) return dataUrl
   134→   134→   134→    ctx.drawImage(img, 0, 0, canvas.width, canvas.height)
   135→   135→   135→    return canvas.toDataURL('image/jpeg', 0.8)
   136→   136→   136→  } catch {
   137→   137→   137→    return dataUrl
   138→   138→   138→  }
   139→   139→   139→}
