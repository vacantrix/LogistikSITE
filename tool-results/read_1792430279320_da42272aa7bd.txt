    82→const PLATE_MAP: Record<string, string> = {
    83→  'А': 'A', 'В': 'B', 'Е': 'E', 'К': 'K', 'М': 'M', 'Н': 'H', 'О': 'O', 'Р': 'P', 'С': 'C', 'Т': 'T', 'У': 'Y', 'Х': 'X',

    85→const normSpace = (s: unknown): string => String(s ?? '').replace(/\s+/g, ' ').trim()
    86→const normFio = (s: unknown): string => normSpace(s).toUpperCase().replace(/Ё/g, 'Е')

    89→/** Номер заявки из ячейки Excel: «1042» → «Z-1042», русская «З» → латинская «Z» */
    90→function normRequestNo(s: unknown): string {
    91→  let v = normSpace(s).toUpperCase()
    92→  if (!v) return ''
    93→  if (/^\d{3,4}$/.test(v)) v = `Z-${v}`
    94→  v = v.replace(/^З/, 'Z')
    95→  return v
    96→}

   113→/** Цвет рамки слота по вердикту нейросети */
   114→function slotBorder(s: SlotState): string {
   115→  if (s.checking) return 'border-blue-300 bg-blue-50/30'
   116→  if (s.error) return 'border-red-300 bg-red-50/40'
   117→  if (!s.verdict) return 'border-slate-200'
   118→  switch (s.verdict.status) {
   119→    case 'PASSED': return 'border-emerald-300 bg-emerald-50/40'
   120→    case 'MANUAL': return 'border-blue-300 bg-blue-50/30'
   121→    case 'REUPLOAD': return 'border-amber-300 bg-amber-50/40'
   122→    case 'REJECTED': return 'border-red-300 bg-red-50/40'
   123→    default: return 'border-slate-200'
   124→  }

   130→  return 'text-amber-600'

   135→  if (s.checking) {

   143→  if (s.error) {
   144→    return (
   145→      <div className="mt-2 flex items-start gap-2 rounded-lg bg-red-50 px-2.5 py-2 text-xs text-red-700">
   146→        <XCircle className="mt-0.5 h-3.5 w-3.5 shrink-0" aria-hidden />
   147→        {s.error}
   148→      </div>
   149→    )
   150→  }
   151→  const v = s.verdict
   152→  if (!v) {
   153→    return <p className="mt-2 text-[11px] leading-snug text-slate-400">{SLOT_META[type].hint}. Нейросеть проверит документ сразу после загрузки.</p>
   154→  }
   155→
   156→  if (v.status === 'REJECTED') {
   157→    return (
   158→      <div className="mt-2 space-y-1.5 rounded-lg border border-red-200 bg-red-50 px-2.5 py-2">
   159→        <div className="flex items-center gap-1.5 text-xs font-semibold text-red-800">
   160→          <XCircle className="h-3.5 w-3.5" aria-hidden />
   161→          Отказ нейросети — замените файл
   162→        </div>
   163→        <p className="text-[11px] leading-snug text-red-700">{v.message}</p>
   164→        <p className="text-[11px] leading-snug text-red-600">Заявка не будет отправлена, пока документ не заменён.</p>
   165→      </div>
   166→    )
   167→  }
   168→  if (v.status === 'REUPLOAD') {
   169→    return (
   170→      <div className="mt-2 space-y-1 rounded-lg border border-amber-200 bg-amber-50 px-2.5 py-2">
   171→        <div className="flex items-center gap-1.5 text-xs font-semibold text-amber-800">
   172→          <AlertTriangle className="h-3.5 w-3.5" aria-hidden />
   173→          <DocStatusBadge status={v.status} />
   174→        </div>
   175→        <p className="text-[11px] leading-snug text-amber-700">{v.message}</p>
   176→      </div>
   177→    )
   178→  }
   179→  if (v.status === 'MANUAL') {
   180→    return (
   181→      <div className="mt-2 space-y-1 rounded-lg border border-blue-200 bg-blue-50 px-2.5 py-2">
   182→        <div className="flex items-center gap-1.5 text-xs font-semibold text-blue-800">
   183→          <Info className="h-3.5 w-3.5" aria-hidden />
   184→          <DocStatusBadge status={v.status} />
   185→        </div>
   186→        <p className="text-[11px] leading-snug text-blue-700">{v.message}</p>
   187→      </div>
   188→    )
   189→  }
   190→  // PASSED: показываем распознанные поля с уверенностью
   191→  const fieldRows = Object.entries(v.fields)
   192→    .filter(([, val]) => val)
   193→    .slice(0, 3)
   194→  return (
   195→    <div className="mt-2 space-y-1 rounded-lg border border-emerald-200 bg-emerald-50 px-2.5 py-2">
   196→      <div className="flex items-center gap-1.5 text-xs font-semibold text-emerald-800">
   197→        <CheckCircle2 className="h-3.5 w-3.5" aria-hidden />
   198→        Вид документа подтверждён · качество {v.quality}
   199→      </div>
   200→      {fieldRows.map(([label, val]) => (
   201→        <div key={label} className="flex items-baseline justify-between gap-2 text-[11px]">
   202→          <span className="text-slate-500">{label}:</span>
   203→          <span className="min-w-0 truncate font-medium text-slate-700">{val}</span>
   204→          <span className={`shrink-0 font-semibold ${confCls(v.confidence[label] ?? 0)}`}>
   205→            {v.confidence[label] != null ? `${Math.round((v.confidence[label] ?? 0) * 100)}%` : ''}
   206→          </span>
   207→        </div>
   208→      ))}
   209→    </div>
   210→  )
   211→}

   336→  /** Разбор Excel: строка = заявка + водитель + тягач (+прицеп), сверка с площадкой и базой аккаунта */
   337→  const parseExcel = async (file: File) => {
   338→    try {
   339→      const wb = XLSX.read(await file.arrayBuffer())
   340→      const ws = wb.Sheets[wb.SheetNames[0]]
   341→      if (!ws) throw new Error('В файле нет листов')
   342→      const sheet = XLSX.utils.sheet_to_json<unknown[]>(ws, { header: 1, defval: '' })
   343→      const grid = sheet.filter((row) => Array.isArray(row) && row.some((c) => normSpace(c)))
   344→      if (grid.length === 0) throw new Error('Лист пуст')
   345→
   346→      // колонки: по заголовкам, иначе позиционно (Заявка · Водитель · Тягач · Прицеп)
   347→      const head = grid[0].map((c) => normSpace(c).toLowerCase())
   348→      const hasHeader = head.some((h) => /заявк|рейс|водител|тягач|прицеп/.test(h))
   349→      const col = (re: RegExp, fallback: number): number => {
   350→        if (hasHeader) {
   351→          const i = head.findIndex((h) => re.test(h))
   352→          if (i !== -1) return i
   353→        }
   354→        return fallback
   355→      }
   356→      const cReq = col(/заявк|рейс/, 0)
   357→      const cDrv = col(/водител|фио/, 1)
   358→      const cTrk = col(/тягач|автомоб/, 2)
   359→      const cTrl = col(/прицеп/, 3)
   360→      const body = hasHeader ? grid.slice(1) : grid
   361→
   362→      const seen = new Set<string>()
   363→      const rows: ImportRow[] = body.map((cells, idx) => {
   364→        const requestNo = normRequestNo(cells[cReq])
   365→        const driverName = normSpace(cells[cDrv])
   366→        const truckPlate = normSpace(cells[cTrk])
   367→        const trailerPlate = normSpace(cells[cTrl])
   368→        const row: ImportRow = {
   369→          line: idx + (hasHeader ? 2 : 1),
   370→          requestNo, driverName, truckPlate, trailerPlate,
   371→          ok: false, message: '',
   372→        }
   373→        if (!requestNo && !driverName && !truckPlate) {
   374→          row.message = 'пустая строка'
   375→          return row
   376→        }
   377→        const byNo = requests?.find((r) => r.number === requestNo)
   378→        if (!byNo) {
   379→          row.message = `заявка «${requestNo || '—'}» не найдена на площадке`
   380→          return row
   381→        }
   382→        if (seen.has(byNo.id)) {
   383→          row.requestId = byNo.id
   384→          row.message = 'дубликат строки — эта заявка уже подаётся в файле'
   385→          return row
   386→        }
   387→        if (byNo.winnerId !== activeCarrierId) {
   388→          row.message = 'рейс не закреплён за вашей компанией'
   389→          return row
   390→        }
   391→        if (byNo.status !== 'AWARDED' && byNo.status !== 'DOCS_REVIEW') {
   392→          row.message = `статус «${REQUEST_STATUS_META[byNo.status].label}» — подача недоступна`
   393→          return row
   394→        }
   395→        const drv = drivers.find((d) => normFio(d.fullName) === normFio(driverName))
   396→        if (!drv) {
   397→          row.message = `водитель «${driverName || '—'}» не найден в базе аккаунта`
   398→          return row
   399→        }
   400→        const trk = trucks.find((t) => normPlate(t.plate) === normPlate(truckPlate))
   401→        if (!trk) {
   402→          row.message = `тягач «${truckPlate || '—'}» не найден в базе аккаунта`
   403→          return row
   404→        }
   405→        let trl: string | undefined
   406→        if (trailerPlate && !/^(—|-|нет|без)$/i.test(trailerPlate)) {
   407→          const found = trailers.find((z) => normPlate(z.plate) === normPlate(trailerPlate))
   408→          if (!found) {
   409→            row.message = `прицеп «${trailerPlate}» не найден в базе аккаунта`
   410→            return row
   411→          }
   412→          trl = found.id
