   128→export default function DirectoriesTab({ activeCarrierId, role, onNavigate }: TabProps) {
   129→  const { toast } = useToast()
   130→  const [carriers, setCarriers] = useState<CarrierDTO[] | null>(null)
   131→  const [cargoTypes, setCargoTypes] = useState<CargoTypeDTO[]>([])
   132→  const [error, setError] = useState<string | null>(null)
   133→  const [tab, setTab] = useState<TabKind>('carriers')
   134→  const [addKind, setAddKind] = useState<Kind | null>(null)
   135→  const [editing, setEditing] = useState<{ kind: Kind; id: string } | null>(null)
   136→  const [form, setForm] = useState<Record<string, string>>({})
   137→  const [saving, setSaving] = useState(false)
   138→  const isLogist = role === 'logist'
   139→
   140→  // перевозчику доступен только его собственный парк — данные конкурентов скрыты
   141→  const visibleCarriers = useMemo(
   142→    () => (isLogist ? (carriers ?? []) : (carriers ?? []).filter((c) => c.id === activeCarrierId)),
   143→    [carriers, isLogist, activeCarrierId],
   144→  )
   145→
   146→  const load = useCallback(async () => {
   147→    try {
   148→      const dir = await api.directory()
   149→      setCarriers(dir.carriers)
   150→      setCargoTypes(dir.cargoTypes)
   151→      setError(null)
   152→    } catch (e) {
   153→      setError(e instanceof Error ? e.message : 'Неизвестная ошибка')
   154→    }
   155→  }, [])
   156→
   157→  useEffect(() => { void load() }, [load])
   158→
   159→  const driversFlat = useMemo(
   160→    () => visibleCarriers.flatMap((c) => c.drivers.map((d) => ({ d, carrierName: c.name }))),
   161→    [visibleCarriers],
   162→  )
   163→  const trucksFlat = useMemo(
   164→    () => visibleCarriers.flatMap((c) => c.trucks.map((t) => ({ t, carrierName: c.name }))),
   165→    [visibleCarriers],
   166→  )
   167→  const trailersFlat = useMemo(
   168→    () => visibleCarriers.flatMap((c) => c.trailers.map((z) => ({ z, carrierName: c.name }))),
   169→    [visibleCarriers],
   170→  )
   171→
   172→  /** Что и когда истекает: все документы с плохими/близкими сроками */
   173→  const expiryItems = useMemo<ExpiryItem[]>(() => {
   174→    const items: ExpiryItem[] = []
   175→    const bad = (i: ReturnType<typeof expiryInfo>) => i.state === 'expired' || i.state === 'warn'
   176→    for (const c of visibleCarriers) {
   177→      for (const d of c.drivers) {
   178→        const v = expiryInfo(d.licenseTill)
   179→        if (bad(v)) items.push({ key: `${d.id}-lic`, tab: 'drivers', icon: User, title: d.fullName, docLabel: 'Водительское удостоверение', info: v, carrierName: c.name })
   180→        const m = expiryInfo(d.medTill)
   181→        if (bad(m)) items.push({ key: `${d.id}-med`, tab: 'drivers', icon: User, title: d.fullName, docLabel: 'Медсправка', info: m, carrierName: c.name })
   182→      }
   183→      for (const t of c.trucks) {
   184→        const v = expiryInfo(t.techTill)
   185→        if (bad(v)) items.push({ key: t.id, tab: 'trucks', icon: Truck, title: `${t.brand} · ${t.plate}`, docLabel: 'Техосмотр', info: v, carrierName: c.name })
   186→      }
   187→      for (const z of c.trailers) {
   188→        const v = expiryInfo(z.techTill)
   189→        if (bad(v)) items.push({ key: z.id, tab: 'trailers', icon: Wrench, title: `${z.type} · ${z.plate}`, docLabel: 'Техосмотр', info: v, carrierName: c.name })
   190→      }
   191→    }
   192→    return items.sort((a, b) => (a.info.days ?? 0) - (b.info.days ?? 0))
   193→  }, [visibleCarriers])
   194→
   195→  const expiredCount = expiryItems.filter((i) => i.info.state === 'expired').length
   196→
   197→  const openAdd = (kind: Kind) => {
   198→    setForm(
   199→      kind === 'cargo'
   200→        ? { truckType: 'Тент', dangerous: 'false' }
   201→        : { carrierId: isLogist ? (carriers?.[0]?.id ?? '') : (activeCarrierId ?? '') },
   202→    )
   203→    setEditing(null)
   204→    setAddKind(kind)
   205→  }
   206→
   207→  const openEdit = (kind: Kind, id: string, record: Record<string, string>) => {
   208→    setForm({ ...record })
   209→    setAddKind(null)
   210→    setEditing({ kind, id })
   211→  }
   212→
   213→  const closeDialog = () => {
   214→    setAddKind(null)
   215→    setEditing(null)
   216→  }
   217→
   218→  const submitAdd = async () => {
   219→    if (!addKind) return
   220→    setSaving(true)
   221→    try {
   222→      await api.createDirectory(addKind, form)
   223→      toast({
   224→        title: 'Запись добавлена в базу',
   225→        description: `${KIND_TITLES[addKind]}: запись создана и доступна при подаче заявки`,
   226→      })
   227→      closeDialog()
   228→      await load()
   229→    } catch (e) {
   230→      toast({
   231→        title: 'Не удалось добавить запись',
   232→        description: e instanceof Error ? e.message : 'Проверьте заполнение полей',
   233→        variant: 'destructive',
   234→      })
   235→    } finally {
   236→      setSaving(false)
   237→    }
   238→  }
   239→
   240→  const submitEdit = async () => {
   241→    if (!editing) return
   242→    setSaving(true)
   243→    try {
   244→      await api.updateDirectory(editing.kind, editing.id, form)
   245→      toast({
   246→        title: 'Запись обновлена',
   247→        description: `${KIND_TITLES[editing.kind]}: данные и сроки документов сохранены`,
   248→      })
   249→      closeDialog()
   250→      await load()
   251→    } catch (e) {
   252→      toast({
   253→        title: 'Не удалось обновить запись',
   254→        description: e instanceof Error ? e.message : 'Проверьте заполнение полей',
   255→        variant: 'destructive',
   256→      })
   257→    } finally {
   258→      setSaving(false)
   259→    }
   260→  }
   261→
   262→  if (carriers === null && !error) return <LoadingBlock label="Загрузка базы аккаунта…" />
   263→  if (error) return <ErrorBlock message={error} onRetry={() => { setError(null); void load() }} />
   264→
   265→  const dialogKind = addKind ?? editing?.kind ?? null
   266→  const addLabel = addKind ? KIND_TITLES[addKind] : editing ? KIND_TITLES[editing.kind] : ''
   267→  // у перевозчика вкладка «Перевозчики» недоступна — ведём на «Водителей»
   268→  const activeTab: TabKind = !isLogist && tab === 'carriers' ? 'drivers' : tab
   269→
   270→  return (
   271→    <div>
   272→      <SectionHeader
   273→        title={isLogist ? 'Справочники' : 'База вашего аккаунта'}
   274→      />
   275→
   276→      {/* Сроки документов: что и когда истекает */}
   277→      <Card className={`mb-4 shadow-sm ${expiredCount > 0 ? 'border-red-200 bg-red-50/40' : expiryItems.length > 0 ? 'border-amber-200 bg-amber-50/40' : 'border-emerald-200 bg-emerald-50/40'}`}>
   278→        <CardContent className="p-4">
   279→          <div className="flex flex-wrap items-center justify-between gap-2">
   280→            <div className="flex items-center gap-2 text-sm font-semibold text-slate-900">
   281→              {expiryItems.length === 0
   282→                ? <CheckCircle2 className="h-4 w-4 text-emerald-600" aria-hidden />
   283→                : <CalendarClock className={`h-4 w-4 ${expiredCount > 0 ? 'text-red-600' : 'text-amber-600'}`} aria-hidden />}
   284→              Сроки документов
   285→              {expiryItems.length > 0 && (
   286→                <Badge variant="outline" className={expiredCount > 0 ? 'border-red-300 bg-red-100 text-red-700' : 'border-amber-300 bg-amber-100 text-amber-700'}>
   287→                  {expiredCount > 0 ? `истекло: ${expiredCount}` : 'требуют внимания'}
   288→                  {expiryItems.length > expiredCount ? ` · истекают: ${expiryItems.length - expiredCount}` : ''}
   289→                </Badge>
   290→              )}
   291→            </div>
   292→            <span className="text-[11px] text-slate-500">подсветка: красный — истёк · янтарный — ≤ 30 дней · серый — ≤ 90 дней</span>
   293→          </div>
   294→
   295→          {expiryItems.length === 0 ? (
   296→            <p className="mt-2 text-sm text-slate-600">
   297→              Все документы базы действительны — истекающих сроков нет.
   298→            </p>
   299→          ) : (
   300→            <ul className={`mt-3 max-h-56 space-y-1.5 overflow-y-auto pr-1 ${THIN_SCROLL}`}>
   301→              {expiryItems.map((item) => {
   302→                const Icon = item.icon
   303→                return (
   304→                  <li key={item.key} className={`flex flex-wrap items-center gap-x-3 gap-y-1 rounded-lg border px-3 py-2 ${item.info.state === 'expired' ? 'border-red-200 bg-white' : 'border-amber-200 bg-white'}`}>
   305→                    <Icon className={`h-4 w-4 shrink-0 ${item.info.state === 'expired' ? 'text-red-500' : 'text-amber-500'}`} aria-hidden />
   306→                    <span className="text-sm font-medium text-slate-800">{item.title}</span>
   307→                    {isLogist && <span className="text-xs text-slate-400">{item.carrierName}</span>}
   308→                    <span className="text-xs text-slate-500">{item.docLabel}</span>
   309→                    <ExpiryBadge date={item.info.date} className="ml-auto" />
   310→                    <button
   311→                      onClick={() => setTab(item.tab)}
   312→                      className="rounded-md px-2 py-1 text-xs font-medium text-blue-700 hover:bg-blue-50"
   313→                    >
   314→                      Открыть
   315→                    </button>
   316→                  </li>
   317→                )
   318→              })}
   319→            </ul>
   320→          )}
   321→        </CardContent>
   322→      </Card>
   323→
   324→      <Tabs value={activeTab} onValueChange={(v) => setTab(v as TabKind)} className="gap-4">
   325→        <TabsList className="flex-wrap">
   326→          {isLogist && (
   327→            <TabsTrigger value="carriers">
   328→              <Users className="h-4 w-4" aria-hidden />
   329→              Перевозчики
   330→            </TabsTrigger>
   331→          )}
   332→          {isLogist && (
   333→            <TabsTrigger value="cargos">
   334→              <Boxes className="h-4 w-4" aria-hidden />
   335→              Грузы
   336→            </TabsTrigger>
   337→          )}
   338→          <TabsTrigger value="drivers">
   339→            <User className="h-4 w-4" aria-hidden />
   340→            Водители
   341→          </TabsTrigger>
   342→          <TabsTrigger value="trucks">
   343→            <Truck className="h-4 w-4" aria-hidden />
   344→            Тягачи (ТС)
   345→          </TabsTrigger>
   346→          <TabsTrigger value="trailers">
   347→            <Wrench className="h-4 w-4" aria-hidden />
   348→            Прицепы
   349→          </TabsTrigger>
   350→        </TabsList>
   351→
   352→        {/* -------- Перевозчики (только логист) -------- */}
   353→        {isLogist && (
   354→        <TabsContent value="carriers">
   355→          <Card className="border-slate-200 shadow-sm">
   356→            <CardContent className="p-4 sm:p-6">
   357→              <div className="mb-3 flex items-center justify-between gap-3">
   358→                <div className="text-sm text-slate-500">Записей: {(carriers ?? []).length}</div>
   359→                <Button size="sm" onClick={() => openAdd('carrier')}>
   360→                  <Plus className="h-4 w-4" aria-hidden />
   361→                  Добавить перевозчика
   362→                </Button>
   363→              </div>
   364→              <div className={`max-h-96 overflow-y-auto overflow-x-auto rounded-lg border border-slate-200 ${THIN_SCROLL}`}>
   365→                <Table>
   366→                  <TableHeader>
   367→                    <TableRow className="bg-slate-50">
   368→                      <TableHead>Название</TableHead>
   369→                      <TableHead>ИНН</TableHead>
   370→                      <TableHead>Договор</TableHead>
   371→                      <TableHead>Доступ</TableHead>
   372→                      <TableHead>Взято заявок</TableHead>
   373→                      <TableHead>Ставок</TableHead>
   374→                      <TableHead>Водители/ТС</TableHead>
   375→                      <TableHead className="w-10"><span className="sr-only">Изменить</span></TableHead>
   376→                    </TableRow>
   377→                  </TableHeader>
   378→                  <TableBody>
   379→                    {(carriers ?? []).map((c) => (
   380→                      <TableRow key={c.id}>
   381→                        <TableCell className="font-medium text-slate-800">{c.name}</TableCell>
   382→                        <TableCell className="font-mono text-xs text-slate-600">{c.inn}</TableCell>
   383→                        <TableCell className="whitespace-nowrap text-slate-600">№{c.contractNo} от {c.contractDate}</TableCell>
   384→                        <TableCell><AccessBadge access={c.access} /></TableCell>
   385→                        <TableCell className="font-semibold text-slate-800">{c.takenCount}</TableCell>
   386→                        <TableCell className="text-slate-600">{c.bidsCount}</TableCell>
   387→                        <TableCell className="whitespace-nowrap text-slate-600">
   388→                          {c.drivers.length} / {c.trucks.length}
   389→                        </TableCell>
   390→                        <TableCell>
   391→                          <Button
   392→                            size="icon"
   393→                            variant="ghost"
   394→                            className="h-8 w-8"
   395→                            aria-label={`Изменить ${c.name}`}
   396→                            onClick={() => openEdit('carrier', c.id, { name: c.name, inn: c.inn, contractNo: c.contractNo, contractDate: c.contractDate, access: String(c.access) })}
   397→                          >
   398→                            <Pencil className="h-3.5 w-3.5" aria-hidden />
   399→                          </Button>
   400→                        </TableCell>
   401→                      </TableRow>
   402→                    ))}
   403→                    {(carriers ?? []).length === 0 && (
   404→                      <TableRow><TableCell colSpan={8} className="text-center text-slate-400">Записей нет</TableCell></TableRow>
   405→                    )}
   406→                  </TableBody>
   407→                </Table>
   408→              </div>
   409→            </CardContent>
   410→          </Card>
   411→        </TabsContent>
   412→        )}
   413→
   414→        {/* -------- Грузы (справочник грузов, только логист) -------- */}
   415→        {isLogist && (
   416→        <TabsContent value="cargos">
   417→          <Card className="border-slate-200 shadow-sm">
   418→            <CardContent className="p-4 sm:p-6">
   419→              <div className="mb-3 flex flex-wrap items-center justify-between gap-3">
   420→                <div className="max-w-xl text-sm text-slate-500">
   421→                  Единый справочник грузов: заявка ссылается на запись отсюда, поэтому один и тот же груз всегда называется одинаково — в площадке, сверке и отчётах.
   422→                </div>
   423→                <Button size="sm" onClick={() => openAdd('cargo')}>
   424→                  <Plus className="h-4 w-4" aria-hidden />
   425→                  Добавить груз
   426→                </Button>
   427→              </div>
   428→              <div className={`max-h-96 overflow-y-auto overflow-x-auto rounded-lg border border-slate-200 ${THIN_SCROLL}`}>
   429→                <Table>
