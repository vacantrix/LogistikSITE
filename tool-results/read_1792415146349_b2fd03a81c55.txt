   105→   105→function confCls(c: number): string {
   106→   106→  if (c >= 0.95) return 'text-emerald-600'
   107→   107→  if (c >= 0.6) return 'text-blue-600'
   108→   108→  return 'text-amber-600'

   111→   111→function ConfChip({ c }: { c: number }) {

   114→   114→}

   189→   189→  const fileInputs = useRef<Partial<Record<DocType, HTMLInputElement | null>>>({})
   190→   190→
   191→   191→  const canRecognize = !!(carrierId && uploads.PASSPORT_PHOTO && uploads.LICENSE)
   192→   192→

   199→   199→    setDraftDriver(null)
   200→   200→    setDraftTruck(null)

   217→   217→
   218→   218→  const setUpload = useCallback((type: DocType, up: Upload | null) => {
   219→   219→    setUploads((prev) => {
   220→   220→      const next = { ...prev }
   221→   221→      if (up) next[type] = up
   222→   222→      else delete next[type]

   224→   224→    })
   225→   225→  }, [])
   226→   226→
   227→   227→  const handleFile = async (type: DocType, file: File) => {

   229→   229→      const dataUri = await fileToCompressedDataUri(file)

   309→   309→        }

   311→   311→        savedRequest = v.request
   312→   312→      }

   315→   315→        title: 'Водитель сохранён в справочнике',

   319→   319→      })

   322→   322→    } catch (e) {
   323→   323→      toast({ title: 'Не удалось сохранить водителя', description: e instanceof Error ? e.message : 'Неизвестная ошибка', variant: 'destructive' })

   325→   325→      setSaving(false)
   326→   326→    }
   327→   327→  }
   328→   328→
   329→   329→  const badDocs: IntakeDocVerdict[] = result?.documents.filter((d) => d.status === 'REUPLOAD' || d.status === 'REJECTED') ?? []
   330→   330→  const canSave = !!(draftDriver?.fullName && draftDriver?.license && carrierId) && !saving
   331→   331→
   332→   332→  return (
   333→   333→    <Dialog open={open} onOpenChange={onOpenChange}>
   334→   334→      <DialogContent className="max-h-[88vh] overflow-y-auto sm:max-w-2xl">
   335→   335→        <DialogHeader>
   336→   336→          <DialogTitle className="flex items-center gap-2">
   337→   337→            <ScanLine className="h-5 w-5 text-blue-700" aria-hidden />
   338→   338→            Новый водитель — по документам, с нейросетью

   344→   344→          </DialogDescription>
   345→   345→        </DialogHeader>
   346→   346→
   347→   347→        {/* индикатор шагов */}
   348→   348→        <ol className="flex flex-wrap items-center gap-1.5 text-xs font-medium" aria-label="Шаги добавления водителя">
   349→   349→          {[
   350→   350→            { n: 1, label: 'Документы', active: step === 'docs' },
   351→   351→            { n: 2, label: 'Нейросеть', active: step === 'scan' },
   352→   352→            { n: 3, label: 'Подтверждение', active: step === 'review' },

   354→   354→            <li key={s.n} className="flex items-center gap-1.5">

   356→   356→              <span className={`flex items-center gap-1.5 rounded-full border px-2.5 py-1 ${s.active ? 'border-blue-200 bg-blue-50 text-blue-800' : 'border-slate-200 bg-slate-50 text-slate-500'}`}>
   357→   357→                <span className={`flex h-4 w-4 items-center justify-center rounded-full text-[10px] ${s.active ? 'bg-blue-700 text-white' : 'bg-slate-300 text-white'}`} aria-hidden>{s.n}</span>
   358→   358→                {s.label}
   359→   359→              </span>
   360→   360→            </li>

   362→   362→        </ol>
   363→   363→
   364→   364→        {/* ——— Шаг 1: документы ——— */}
   365→   365→        {step === 'docs' && (
