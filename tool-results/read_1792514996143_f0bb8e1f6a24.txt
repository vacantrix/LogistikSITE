   122→export default function DirectoriesTab({ activeCarrierId, role, onNavigate }: TabProps) {
   123→  const { toast } = useToast()
   124→  const [carriers, setCarriers] = useState<CarrierDTO[] | null>(null)
   125→  const [error, setError] = useState<string | null>(null)
   126→  const [tab, setTab] = useState<TabKind>('carriers')
   127→  const [addKind, setAddKind] = useState<Kind | null>(null)
   128→  const [editing, setEditing] = useState<{ kind: Kind; id: string } | null>(null)
   129→  const [form, setForm] = useState<Record<string, string>>({})
   130→  const [saving, setSaving] = useState(false)
   131→  const isLogist = role === 'logist'
   132→
   133→  // перевозчику доступен только его собственный парк — данные конкурентов скрыты
   134→  const visibleCarriers = useMemo(
   135→    () => (isLogist ? (carriers ?? []) : (carriers ?? []).filter((c) => c.id === activeCarrierId)),
   136→    [carriers, isLogist, activeCarrierId],
   137→  )
   138→
   139→  const load = useCallback(async () => {
   140→    try {
   141→      const dir = await api.directory()
   142→      setCarriers(dir.carriers)
   143→      setError(null)
   144→    } catch (e) {
   145→      setError(e instanceof Error ? e.message : 'Неизвестная ошибка')
   146→    }
   147→  }, [])
   148→
   149→  useEffect(() => { void load() }, [load])
   150→
   151→  const driversFlat = useMemo(
   152→    () => visibleCarriers.flatMap((c) => c.drivers.map((d) => ({ d, carrierName: c.name }))),
   153→    [visibleCarriers],
   154→  )
   155→  const trucksFlat = useMemo(
   156→    () => visibleCarriers.flatMap((c) => c.trucks.map((t) => ({ t, carrierName: c.name }))),
   157→    [visibleCarriers],
   158→  )
   159→  const trailersFlat = useMemo(
   160→    () => visibleCarriers.flatMap((c) => c.trailers.map((z) => ({ z, carrierName: c.name }))),
   161→    [visibleCarriers],
   162→  )
   163→
   164→  /** Что и когда истекает: все документы с плохими/близкими сроками */
   165→  const expiryItems = useMemo<ExpiryItem[]>(() => {
   166→    const items: ExpiryItem[] = []
   167→    const bad = (i: ReturnType<typeof expiryInfo>) => i.state === 'expired' || i.state === 'warn'
   168→    for (const c of visibleCarriers) {
   169→      for (const d of c.drivers) {
   170→        const v = expiryInfo(d.licenseTill)
   171→        if (bad(v)) items.push({ key: `${d.id}-lic`, tab: 'drivers', icon: User, title: d.fullName, docLabel: 'Водительское удостоверение', info: v, carrierName: c.name })
   172→        const m = expiryInfo(d.medTill)
   173→        if (bad(m)) items.push({ key: `${d.id}-med`, tab: 'drivers', icon: User, title: d.fullName, docLabel: 'Медсправка', info: m, carrierName: c.name })
   174→      }
   175→      for (const t of c.trucks) {
   176→        const v = expiryInfo(t.techTill)
   177→        if (bad(v)) items.push({ key: t.id, tab: 'trucks', icon: Truck, title: `${t.brand} · ${t.plate}`, docLabel: 'Техосмотр', info: v, carrierName: c.name })
   178→      }
   179→      for (const z of c.trailers) {
   180→        const v = expiryInfo(z.techTill)
   181→        if (bad(v)) items.push({ key: z.id, tab: 'trailers', icon: Wrench, title: `${z.type} · ${z.plate}`, docLabel: 'Техосмотр', info: v, carrierName: c.name })
   182→      }
   183→    }
   184→    return items.sort((a, b) => (a.info.days ?? 0) - (b.info.days ?? 0))
   185→  }, [visibleCarriers])
   186→
   187→  const expiredCount = expiryItems.filter((i) => i.info.state === 'expired').length
   188→
   189→  const openAdd = (kind: Kind) => {
   190→    setForm({ carrierId: isLogist ? (carriers?.[0]?.id ?? '') : (activeCarrierId ?? '') })
   191→    setEditing(null)
   192→    setAddKind(kind)
   193→  }
   194→
   195→  const openEdit = (kind: Kind, id: string, record: Record<string, string>) => {
   196→    setForm({ ...record })
   197→    setAddKind(null)
   198→    setEditing({ kind, id })
   199→  }
   200→
   201→  const closeDialog = () => {
   202→    setAddKind(null)
   203→    setEditing(null)
   204→  }
   205→
   206→  const submitAdd = async () => {
   207→    if (!addKind) return
   208→    setSaving(true)
   209→    try {
   210→      await api.createDirectory(addKind, form)
   211→      toast({
   212→        title: 'Запись добавлена в базу',
   213→        description: `${KIND_TITLES[addKind]}: запись создана и доступна при подаче заявки`,
   214→      })
   215→      closeDialog()
   216→      await load()
   217→    } catch (e) {
   218→      toast({
   219→        title: 'Не удалось добавить запись',
   220→        description: e instanceof Error ? e.message : 'Проверьте заполнение полей',
   221→        variant: 'destructive',
   222→      })
   223→    } finally {
   224→      setSaving(false)
   225→    }
   226→  }
   227→
   228→  const submitEdit = async () => {
   229→    if (!editing) return
   230→    setSaving(true)
   231→    try {
   232→      await api.updateDirectory(editing.kind, editing.id, form)
   233→      toast({
   234→        title: 'Запись обновлена',
   235→        description: `${KIND_TITLES[editing.kind]}: данные и сроки документов сохранены`,
   236→      })
   237→      closeDialog()
   238→      await load()
   239→    } catch (e) {
   240→      toast({
   241→        title: 'Не удалось обновить запись',
   242→        description: e instanceof Error ? e.message : 'Проверьте заполнение полей',
   243→        variant: 'destructive',
   244→      })
   245→    } finally {
   246→      setSaving(false)
   247→    }
   248→  }
   249→
   250→  if (carriers === null && !error) return <LoadingBlock label="Загрузка базы аккаунта…" />
   251→  if (error) return <ErrorBlock message={error} onRetry={() => { setError(null); void load() }} />
   252→
   253→  const dialogKind = addKind ?? editing?.kind ?? null
   254→  const addLabel = addKind ? KIND_TITLES[addKind] : editing ? KIND_TITLES[editing.kind] : ''
   255→  // у перевозчика вкладка «Перевозчики» недоступна — ведём на «Водителей»
   256→  const activeTab: TabKind = !isLogist && tab === 'carriers' ? 'drivers' : tab
   257→
   258→  return (
   259→    <div>
   260→      <SectionHeader
   261→        title={isLogist ? 'Справочники' : 'База вашего аккаунта'}
   262→      />
   263→
   264→      {/* Сроки документов: что и когда истекает */}
   265→      <Card className={`mb-4 shadow-sm ${expiredCount > 0 ? 'border-red-200 bg-red-50/40' : expiryItems.length > 0 ? 'border-amber-200 bg-amber-50/40' : 'border-emerald-200 bg-emerald-50/40'}`}>
   266→        <CardContent className="p-4">
   267→          <div className="flex flex-wrap items-center justify-between gap-2">
   268→            <div className="flex items-center gap-2 text-sm font-semibold text-slate-900">
   269→              {expiryItems.length === 0
   270→                ? <CheckCircle2 className="h-4 w-4 text-emerald-600" aria-hidden />
   271→                : <CalendarClock className={`h-4 w-4 ${expiredCount > 0 ? 'text-red-600' : 'text-amber-600'}`} aria-hidden />}
   272→              Сроки документов
   273→              {expiryItems.length > 0 && (
   274→                <Badge variant="outline" className={expiredCount > 0 ? 'border-red-300 bg-red-100 text-red-700' : 'border-amber-300 bg-amber-100 text-amber-700'}>
   275→                  {expiredCount > 0 ? `истекло: ${expiredCount}` : 'требуют внимания'}
   276→                  {expiryItems.length > expiredCount ? ` · истекают: ${expiryItems.length - expiredCount}` : ''}
   277→                </Badge>
   278→              )}
   279→            </div>
   280→            <span className="text-[11px] text-slate-500">подсветка: красный — истёк · янтарный — ≤ 30 дней · серый — ≤ 90 дней</span>
   281→          </div>
   282→
   283→          {expiryItems.length === 0 ? (
   284→            <p className="mt-2 text-sm text-slate-600">
   285→              Все документы базы действительны — истекающих сроков нет.
   286→            </p>
   287→          ) : (
   288→            <ul className={`mt-3 max-h-56 space-y-1.5 overflow-y-auto pr-1 ${THIN_SCROLL}`}>
   289→              {expiryItems.map((item) => {
   290→                const Icon = item.icon
   291→                return (
   292→                  <li key={item.key} className={`flex flex-wrap items-center gap-x-3 gap-y-1 rounded-lg border px-3 py-2 ${item.info.state === 'expired' ? 'border-red-200 bg-white' : 'border-amber-200 bg-white'}`}>
   293→                    <Icon className={`h-4 w-4 shrink-0 ${item.info.state === 'expired' ? 'text-red-500' : 'text-amber-500'}`} aria-hidden />
   294→                    <span className="text-sm font-medium text-slate-800">{item.title}</span>
   295→                    {isLogist && <span className="text-xs text-slate-400">{item.carrierName}</span>}
   296→                    <span className="text-xs text-slate-500">{item.docLabel}</span>
   297→                    <ExpiryBadge date={item.info.date} className="ml-auto" />
   298→                    <button
   299→                      onClick={() => setTab(item.tab)}
   300→                      className="rounded-md px-2 py-1 text-xs font-medium text-blue-700 hover:bg-blue-50"
   301→                    >
   302→                      Открыть
   303→                    </button>
   304→                  </li>
   305→                )
   306→              })}
   307→            </ul>
   308→          )}
   309→        </CardContent>
   310→      </Card>
   311→
   312→      <Tabs value={activeTab} onValueChange={(v) => setTab(v as TabKind)} className="gap-4">
   313→        <TabsList className="flex-wrap">
   314→          {isLogist && (
   315→            <TabsTrigger value="carriers">
   316→              <Users className="h-4 w-4" aria-hidden />
   317→              Перевозчики
   318→            </TabsTrigger>
   319→          )}
   320→          <TabsTrigger value="drivers">
   321→            <User className="h-4 w-4" aria-hidden />
   322→            Водители
   323→          </TabsTrigger>
   324→          <TabsTrigger value="trucks">
   325→            <Truck className="h-4 w-4" aria-hidden />
   326→            Тягачи (ТС)
   327→          </TabsTrigger>
   328→          <TabsTrigger value="trailers">
   329→            <Wrench className="h-4 w-4" aria-hidden />
   330→            Прицепы
   331→          </TabsTrigger>
   332→        </TabsList>
   333→
   334→        {/* -------- Перевозчики (только логист) -------- */}
   335→        {isLogist && (
   336→        <TabsContent value="carriers">
   337→          <Card className="border-slate-200 shadow-sm">
   338→            <CardContent className="p-4 sm:p-6">
   339→              <div className="mb-3 flex items-center justify-between gap-3">
   340→                <div className="text-sm text-slate-500">Записей: {(carriers ?? []).length}</div>
   341→                <Button size="sm" onClick={() => openAdd('carrier')}>
   342→                  <Plus className="h-4 w-4" aria-hidden />
   343→                  Добавить перевозчика
   344→                </Button>
   345→              </div>
   346→              <div className={`max-h-96 overflow-y-auto overflow-x-auto rounded-lg border border-slate-200 ${THIN_SCROLL}`}>
   347→                <Table>
   348→                  <TableHeader>
   349→                    <TableRow className="bg-slate-50">
   350→                      <TableHead>Название</TableHead>
   351→                      <TableHead>ИНН</TableHead>
   352→                      <TableHead>Договор</TableHead>
   353→                      <TableHead>Доступ</TableHead>
   354→                      <TableHead>Взято заявок</TableHead>
   355→                      <TableHead>Ставок</TableHead>
   356→                      <TableHead>Водители/ТС</TableHead>
   357→                      <TableHead className="w-10"><span className="sr-only">Изменить</span></TableHead>
   358→                    </TableRow>
   359→                  </TableHeader>
   360→                  <TableBody>
   361→                    {(carriers ?? []).map((c) => (
   362→                      <TableRow key={c.id}>
   363→                        <TableCell className="font-medium text-slate-800">{c.name}</TableCell>
   364→                        <TableCell className="font-mono text-xs text-slate-600">{c.inn}</TableCell>
   365→                        <TableCell className="whitespace-nowrap text-slate-600">№{c.contractNo} от {c.contractDate}</TableCell>
   366→                        <TableCell><AccessBadge access={c.access} /></TableCell>
   367→                        <TableCell className="font-semibold text-slate-800">{c.takenCount}</TableCell>
   368→                        <TableCell className="text-slate-600">{c.bidsCount}</TableCell>
   369→                        <TableCell className="whitespace-nowrap text-slate-600">
   370→                          {c.drivers.length} / {c.trucks.length}
   371→                        </TableCell>
   372→                        <TableCell>
   373→                          <Button
   374→                            size="icon"
   375→                            variant="ghost"
   376→                            className="h-8 w-8"
   377→                            aria-label={`Изменить ${c.name}`}
   378→                            onClick={() => openEdit('carrier', c.id, { name: c.name, inn: c.inn, contractNo: c.contractNo, contractDate: c.contractDate, access: String(c.access) })}
   379→                          >
   380→                            <Pencil className="h-3.5 w-3.5" aria-hidden />
   381→                          </Button>
   382→                        </TableCell>
   383→                      </TableRow>
   384→                    ))}
   385→                    {(carriers ?? []).length === 0 && (
   386→                      <TableRow><TableCell colSpan={8} className="text-center text-slate-400">Записей нет</TableCell></TableRow>
   387→                    )}
   388→                  </TableBody>
   389→                </Table>
   390→              </div>
   391→            </CardContent>
   392→          </Card>
   393→        </TabsContent>
   394→        )}
   395→
   396→        {/* -------- Водители -------- */}
   397→        <TabsContent value="drivers">
   398→          <Card className="border-slate-200 shadow-sm">
   399→            <CardContent className="p-4 sm:p-6">
   400→              {!isLogist && (
   401→                <div className="mb-3 flex flex-wrap items-center justify-between gap-2 rounded-xl border border-blue-200 bg-blue-50/70 p-3">
   402→                  <p className="max-w-xl text-xs leading-snug text-slate-600">
   403→                    <b className="text-slate-800">Новый водитель?</b> Добавляйте по документам во вкладке «Подать заявку» — нейросеть сама прочитает паспорт, ВУ и СТС и заполнит карточку.
   404→                  </p>
   405→                  <Button size="sm" variant="outline" className="gap-1.5" onClick={() => onNavigate('apply')}>
   406→                    Подать заявку
   407→                    <ArrowRight className="h-3.5 w-3.5" aria-hidden />
   408→                  </Button>
   409→                </div>
   410→              )}
   411→              <div className="mb-3 flex items-center justify-between gap-3">
   412→                <div className="text-sm text-slate-500">Записей: {driversFlat.length}</div>
   413→                <Button size="sm" variant="outline" className="gap-1.5" onClick={() => openAdd('driver')}>
   414→                  <Plus className="h-4 w-4" aria-hidden />
   415→                  Добавить вручную
   416→                </Button>
   417→              </div>
   418→              <div className={`max-h-96 overflow-y-auto overflow-x-auto rounded-lg border border-slate-200 ${THIN_SCROLL}`}>
   419→                <Table>
   420→                  <TableHeader>
   421→                    <TableRow className="bg-slate-50">
   422→                      <TableHead>ФИО</TableHead>

   426→                      <TableHead>ВУ действует до</TableHead>
   427→                      <TableHead>Медсправка до</TableHead>
   428→                      <TableHead>Категории</TableHead>
   429→                      <TableHead>Добавлен</TableHead>
   430→                      {!isLogist && <TableHead className="w-10"><span className="sr-only">Изменить</span></TableHead>}
   431→                    </TableRow>
   432→                  </TableHeader>
   433→                  <TableBody>
   434→                    {driversFlat.map(({ d, carrierName }) => (
   435→                      <DriverRow key={d.id} d={d} carrierName={carrierName} showCarrier={isLogist} onEdit={() => openEdit('driver', d.id, {
   436→                        fullName: d.fullName, birthDate: d.birthDate, passport: d.passport, license: d.license,
   437→                        licenseTill: d.licenseTill === '—' ? '' : d.licenseTill, medTill: d.medTill === '—' ? '' : d.medTill,
   438→                        categories: d.categories,
   439→                      })} />
   440→                    ))}
   441→                    {driversFlat.length === 0 && (
   442→                      <TableRow><TableCell colSpan={isLogist ? 8 : 9} className="text-center text-slate-400">Записей нет</TableCell></TableRow>
   443→                    )}
   444→                  </TableBody>
   445→                </Table>
   446→              </div>
   447→            </CardContent>
   448→          </Card>
   449→        </TabsContent>
   450→
   451→        {/* -------- Тягачи -------- */}
   452→        <TabsContent value="trucks">
   453→          <Card className="border-slate-200 shadow-sm">
   454→            <CardContent className="p-4 sm:p-6">
   455→              <div className="mb-3 flex items-center justify-between gap-3">
   456→                <div className="text-sm text-slate-500">Записей: {trucksFlat.length}</div>
   457→                <Button size="sm" variant="outline" className="gap-1.5" onClick={() => openAdd('truck')}>
   458→                  <Plus className="h-4 w-4" aria-hidden />
   459→                  Добавить тягач
   460→                </Button>
   461→              </div>
   462→              <div className={`max-h-96 overflow-y-auto overflow-x-auto rounded-lg border border-slate-200 ${THIN_SCROLL}`}>

   464→                  <TableHeader>

   469→                      <TableHead>СТС</TableHead>
   470→                      <TableHead>Техосмотр до</TableHead>

   472→                      <TableHead>Добавлен</TableHead>
   473→                      {!isLogist && <TableHead className="w-10"><span className="sr-only">Изменить</span></TableHead>}
   474→                    </TableRow>
   475→                  </TableHeader>
   476→                  <TableBody>
   477→                    {trucksFlat.map(({ t, carrierName }) => (
   478→                      <TruckRow key={t.id} t={t} carrierName={carrierName} showCarrier={isLogist} onEdit={() => openEdit('truck', t.id, {
   479→                        brand: t.brand, plate: t.plate, vin: t.vin, sts: t.sts,
   480→                        techTill: t.techTill === '—' ? '' : t.techTill, owner: t.owner,
   481→                      })} />
   482→                    ))}
   483→                    {trucksFlat.length === 0 && (
   484→                      <TableRow><TableCell colSpan={isLogist ? 7 : 8} className="text-center text-slate-400">Записей нет</TableCell></TableRow>
   485→                    )}
   486→                  </TableBody>
   487→                </Table>
   488→              </div>
   489→            </CardContent>
   490→          </Card>
   491→        </TabsContent>
   492→
   493→        {/* -------- Прицепы -------- */}
   494→        <TabsContent value="trailers">
   495→          <Card className="border-slate-200 shadow-sm">
   496→            <CardContent className="p-4 sm:p-6">
   497→              <div className="mb-3 flex items-center justify-between gap-3">
   498→                <div className="text-sm text-slate-500">Записей: {trailersFlat.length}</div>

   502→                </Button>

   526→                </Table>
   527→              </div>
   528→            </CardContent>
   529→          </Card>
   530→        </TabsContent>
   531→      </Tabs>
   532→
   533→      {/* диалог добавления / изменения записи */}
   534→      <Dialog open={!!dialogKind} onOpenChange={(o) => { if (!o) closeDialog() }}>
   535→        <DialogContent className="max-h-[85vh] overflow-y-auto sm:max-w-lg">
   536→          <DialogHeader>
   537→            <DialogTitle>{addKind ? 'Добавить запись' : 'Изменить запись'} — {addLabel}</DialogTitle>
   538→            <DialogDescription>
   539→              {addKind
   540→                ? 'Запись появится в базе аккаунта и станет доступна при подаче заявки.'
   541→                : 'Обновите данные и сроки действия документов — подсветка сроков пересчитается автоматически.'}
   542→            </DialogDescription>
   543→          </DialogHeader>
   544→
   545→          <div className="space-y-3">
   546→            {addKind === 'driver' || addKind === 'truck' || addKind === 'trailer' ? (
   547→              <div className="space-y-1.5">
   548→                <Label>Перевозчик</Label>
   549→                <Select value={form.carrierId ?? ''} onValueChange={(v) => setForm((f) => ({ ...f, carrierId: v }))} disabled={!!editing}>
   550→                  <SelectTrigger aria-label="Выбор перевозчика">
   551→                    <SelectValue placeholder="Выберите перевозчика" />
   552→                  </SelectTrigger>
   553→                  <SelectContent>
   554→                    {(carriers ?? []).map((c) => (
   555→                      <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   556→                    ))}
   557→                  </SelectContent>
   558→                </Select>
   559→              </div>
   560→            ) : null}
   561→
   562→            {dialogKind && FORM_FIELDS[dialogKind].map((field) => (
   563→              <div key={field.key} className="space-y-1.5">
   564→                <Label htmlFor={`add-${field.key}`}>{field.label}</Label>
   565→                <Input
   566→                  id={`add-${field.key}`}
   567→                  placeholder={field.placeholder}
   568→                  value={form[field.key] ?? ''}
   569→                  onChange={(e) => setForm((f) => ({ ...f, [field.key]: e.target.value }))}
   570→                />
   571→              </div>
   572→            ))}
   573→
   574→            {editing?.kind === 'carrier' && (
   575→              <div className="space-y-1.5">
   576→                <Label htmlFor="add-access">Доступ к площадке</Label>
   577→                <Select value={form.access ?? 'true'} onValueChange={(v) => setForm((f) => ({ ...f, access: v }))}>
   578→                  <SelectTrigger id="add-access" aria-label="Доступ к площадке">
   579→                    <SelectValue />
   580→                  </SelectTrigger>
   581→                  <SelectContent>
   582→                    <SelectItem value="true">Доступ открыт</SelectItem>
   583→                    <SelectItem value="false">Закрыт</SelectItem>
   584→                  </SelectContent>
   585→                </Select>
   586→              </div>
   587→            )}
   588→          </div>
   589→
   590→          <DialogFooter>
   591→            <Button variant="outline" onClick={closeDialog}>Отмена</Button>
   592→            <Button disabled={saving} onClick={() => void (addKind ? submitAdd() : submitEdit())}>
   593→              {saving ? 'Сохранение…' : addKind ? 'Добавить' : 'Сохранить'}
   594→            </Button>
   595→          </DialogFooter>
   596→        </DialogContent>
   597→      </Dialog>
   598→    </div>
   599→  )
   600→}

   604→function DriverRow({ d, carrierName, showCarrier, onEdit }: { d: DriverDTO; carrierName: string; showCarrier: boolean; onEdit: () => void }) {
   605→  const lic = expiryInfo(d.licenseTill)
   606→  const med = expiryInfo(d.medTill)
   607→  const tint = rowTint([lic.state, med.state] as ExpiryState[])
   608→  return (
   609→    <TableRow className={tint}>
   610→      <TableCell className="font-medium text-slate-800">{d.fullName}</TableCell>
   611→      <TableCell className="whitespace-nowrap text-slate-600">{d.birthDate}</TableCell>
   612→      <TableCell className="whitespace-nowrap font-mono text-xs text-slate-600">{d.passport}</TableCell>
   613→      <TableCell className="whitespace-nowrap font-mono text-xs text-slate-600">{d.license}</TableCell>
   614→      <TableCell><ExpiryBadge date={d.licenseTill} /></TableCell>
   615→      <TableCell><ExpiryBadge date={d.medTill} /></TableCell>
   616→      <TableCell>
   617→        <div className="flex flex-wrap gap-1">
   618→          {d.categories.split(',').map((cat) => cat.trim()).filter(Boolean).map((cat) => (
   619→            <Badge key={cat} variant="outline" className="border-slate-300 bg-slate-50 text-slate-700">{cat}</Badge>
   620→          ))}
   621→        </div>
   622→      </TableCell>
   623→      <TableCell className="whitespace-nowrap text-xs text-slate-400">{fmtCreated(d.createdAt)}</TableCell>
   624→      {showCarrier && <TableCell className="text-slate-600">{carrierName}</TableCell>}
   625→      {!showCarrier && (
   626→        <TableCell>
   627→          <Button size="icon" variant="ghost" className="h-8 w-8" aria-label={`Изменить ${d.fullName}`} onClick={onEdit}>
   628→            <Pencil className="h-3.5 w-3.5" aria-hidden />
   629→          </Button>
   630→        </TableCell>
   631→      )}
   632→    </TableRow>
   633→  )
