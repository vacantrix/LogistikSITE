    93→/** Текст-подсказка для агента по статусу заявки */
    94→function agentLine(r: RequestDTO): { text: string; cls: string } {
    95→  switch (r.status) {
    96→    case 'AWARDED':
    97→      return { text: 'Укажите водителя и ТС, загрузите документы', cls: 'border-amber-200 bg-amber-50 text-amber-900' }
    98→    case 'DOCS_REVIEW': {
    99→      const bad = r.documents.some((d) => d.status === 'REUPLOAD' || d.status === 'REJECTED')
   100→      return bad
   101→        ? { text: 'Требует исправления: причина по каждому документу, догрузка — в ту же заявку', cls: 'border-amber-300 bg-amber-50 text-amber-900' }
   102→        : { text: 'Нейросеть проверяет документы', cls: 'border-blue-200 bg-blue-50 text-blue-900' }
   103→    }
   104→    case 'AT_LOGIST':
   105→      return { text: 'Комплект у логиста — идёт сверка фото с полями', cls: 'border-blue-200 bg-blue-50 text-blue-900' }
   106→    case 'CONFIRMED':
   107→      return { text: 'Принята — данные переданы в 1С', cls: 'border-emerald-200 bg-emerald-50 text-emerald-900' }
   108→    default:
   109→      return { text: 'Рейс закрыт', cls: 'border-slate-300 bg-slate-100 text-slate-600' }
   110→  }
   111→}

   113→/** Цвет процента уверенности распознавания */
   114→function confClass(c: number): string {
   115→  if (c >= 0.95) return 'text-emerald-600'
   116→  if (c >= 0.6) return 'text-blue-600'
   117→  return 'text-amber-600'
   118→}

   120→/** Цвет списка замечаний по статусу документа */
   121→function issueClass(status: DocStatus): string {
   122→  if (status === 'REJECTED') return 'text-red-600'
   123→  if (status === 'REUPLOAD') return 'text-amber-600'
   124→  if (status === 'MANUAL') return 'text-blue-600'
   125→  return 'text-slate-500'
   126→}
