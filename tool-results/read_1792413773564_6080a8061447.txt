    20→  AlertDialogHeader,
    21→  AlertDialogTitle,
    22→} from '@/components/ui/alert-dialog'
    23→import { Badge } from '@/components/ui/badge'

    26→import {

    33→} from '@/components/ui/dialog'

    36→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'

    52→const THIN_SCROLL = '[&::-webkit-scrollbar]:w-1.5 [&::-webkit-scrollbar-thumb]:rounded-full [&::-webkit-scrollbar-thumb]:bg-slate-300'
    53→
    54→type ModeFilter = 'ALL' | RequestMode
    55→
    56→const MODE_FILTERS: { value: ModeFilter; label: string }[] = [
    57→  { value: 'ALL', label: 'Все режимы' },
    58→  { value: 'FIXED', label: 'Фиксированная ставка' },
    59→  { value: 'AUCTION', label: 'Торги' },
    60→  { value: 'CLOSED', label: 'Закрытые ставки' },
    61→]
    62→
    63→/** Ставка в таблице: по режиму заявки */
    64→function RateCell({ request }: { request: RequestDTO }) {
    65→  if (request.mode === 'FIXED') {
    66→    return <span className="font-semibold text-slate-900">{money(request.startRate)}</span>
    67→  }
    68→  if (request.mode === 'CLOSED') {
    69→    return (
    70→      <div>
    71→        <div className="flex items-center gap-1 font-semibold text-slate-900">
    72→          <Lock className="h-3.5 w-3.5 text-slate-400" aria-hidden />
    73→          Скрыта
    74→        </div>
    75→        <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    76→      </div>
    77→    )
    78→  }
    79→  if (request.bestRate == null) {
    80→    return (
    81→      <div>
    82→        <div className="text-sm font-medium text-slate-500">Ставок нет</div>
    83→        <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    84→      </div>
    85→    )
    86→  }
    87→  return (
    88→    <div>
    89→      <div className="font-semibold text-slate-900">{money(request.bestRate)}</div>
    90→      <div className="text-xs text-muted-foreground">старт {money(request.startRate)}</div>
    91→    </div>
    92→  )
    93→}
    94→
    95→/** Обратный отсчёт до окончания приёма ставок (без hydration mismatch) */
    96→function DeadlineCell({ request, now }: { request: RequestDTO; now: number | null }) {
    97→  if (request.mode === 'FIXED' || !request.deadline) return <span className="text-slate-400">—</span>
    98→  if (now === null) return <span className="text-slate-400">—</span>
    99→  const tl = timeLeft(request.deadline)
   100→  if (!tl) return <span className="text-slate-400">—</span>
   101→  return (
   102→    <span className={`tabular-nums ${tl.urgent ? 'font-semibold text-red-600' : 'text-slate-700'}`}>
   103→      {tl.text}
   104→    </span>
   105→  )
   106→}
   107→
   108→export default function BoardTab({ activeCarrierId }: TabProps) {
   109→  const { toast } = useToast()
   110→  const [requests, setRequests] = useState<RequestDTO[] | null>(null)
   111→  const [loading, setLoading] = useState(true)
   112→  const [refreshing, setRefreshing] = useState(false)
   113→  const [error, setError] = useState<string | null>(null)
   114→  const [q, setQ] = useState('')
   115→  const [mode, setMode] = useState<ModeFilter>('ALL')
   116→  const [now, setNow] = useState<number | null>(null)
   117→
   118→  // диалоги: держим только id, свежие данные берём из списка
   119→  const [detailId, setDetailId] = useState<string | null>(null)
   120→  const [bidId, setBidId] = useState<string | null>(null)
   121→  const [bidAmount, setBidAmount] = useState('')
   122→  const [takeId, setTakeId] = useState<string | null>(null)
   123→  const [busyId, setBusyId] = useState<string | null>(null)
   124→
   125→  const load = useCallback(async (silent = false) => {
   126→    if (!silent) setRefreshing(true)
   127→    try {
   128→      const d = await api.listRequests()
   129→      setRequests(d.requests)
   130→      setError(null)
   131→    } catch (e) {
   132→      setError(e instanceof Error ? e.message : 'Неизвестная ошибка')
   133→    } finally {
   134→      setLoading(false)
   135→      if (!silent) setRefreshing(false)
   136→    }
   137→  }, [])
   138→
   139→  // первичная загрузка + автообновление каждые 15 секунд
   140→  useEffect(() => {
   141→    void load()
   142→    const t = setInterval(() => void load(true), 15_000)
   143→    return () => clearInterval(t)
   144→  }, [load])
   145→
   146→  // таймер для обратного отсчёта — только после монтирования
   147→  useEffect(() => {
   148→    setNow(Date.now())
   149→    const t = setInterval(() => setNow(Date.now()), 1000)
   150→    return () => clearInterval(t)
   151→  }, [])
   152→
   153→  const applyRequest = useCallback((updated: RequestDTO) => {
   154→    setRequests((prev) => {
   155→      const list = prev ?? []
   156→      const i = list.findIndex((r) => r.id === updated.id)
   157→      if (i === -1) return [updated, ...list]
   158→      const next = [...list]
   159→      next[i] = updated
   160→      return next
   161→    })
   162→  }, [])
   163→
   164→  const filtered = useMemo(() => {
   165→    const list = requests ?? []
   166→    const needle = q.trim().toLowerCase()
   167→    return list.filter((r) => {
   168→      if (mode !== 'ALL' && r.mode !== mode) return false
   169→      if (!needle) return true
   170→      return [r.number, r.cargo, r.fromPoint, r.toPoint].some((v) => v.toLowerCase().includes(needle))
   171→    })
   172→  }, [requests, q, mode])
   173→
   174→  const openCount = (requests ?? []).filter((r) => r.status === 'PUBLISHED').length
   175→
   176→  const detailReq = (requests ?? []).find((r) => r.id === detailId) ?? null
   177→  const bidReq = (requests ?? []).find((r) => r.id === bidId) ?? null
   178→  const takeReq = (requests ?? []).find((r) => r.id === takeId) ?? null
   179→
   180→  const closeBid = () => {
   181→    setBidId(null)
   182→    setBidAmount('')
   183→  }
   184→
   185→  const doBid = async () => {
   186→    if (!bidReq || !activeCarrierId) return
   187→    const amount = Math.round(Number(bidAmount.replace(/\s/g, '').replace(',', '.')))
   188→    if (!amount || amount <= 0) {
   189→      toast({ title: 'Ошибка', description: 'Укажите ставку в рублях', variant: 'destructive' })
   190→      return
   191→    }
   192→    if (bidReq.mode === 'AUCTION') {
   193→      const floor = bidReq.bestRate ?? bidReq.startRate
   194→      if (amount >= floor) {
   195→        toast({ title: 'Ошибка', description: `Ставка должна быть ниже текущей лучшей (${money(floor)})`, variant: 'destructive' })
   196→        return
   197→      }
   198→    }
   199→    setBusyId(bidReq.id)
   200→    try {
   201→      const { request } = await api.action(bidReq.id, 'bid', { carrierId: activeCarrierId, amount })
   202→      applyRequest(request)
   203→      toast({ title: 'Ставка принята', description: `${request.number} — ${money(amount)}${request.mode === 'CLOSED' ? ' · закрытая ставка' : ''}` })
   204→      closeBid()
   205→    } catch (e) {
   206→      toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Не удалось подать ставку', variant: 'destructive' })
   207→    } finally {
   208→      setBusyId(null)
   209→    }
   210→  }
   211→
   212→  const doTake = async () => {
   213→    if (!takeReq || !activeCarrierId) return
   214→    setBusyId(takeReq.id)
   215→    try {
   216→      const { request } = await api.action(takeReq.id, 'take', { carrierId: activeCarrierId })
   217→      applyRequest(request)
   218→      toast({ title: 'Заявка закреплена за вами', description: `${request.number} — фиксированная ставка ${money(request.startRate)}` })
   219→      setTakeId(null)
   220→    } catch (e) {
   221→      toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Не удалось взять заявку', variant: 'destructive' })
   222→    } finally {
   223→      setBusyId(null)
   224→    }
   225→  }
   226→
   227→  const sortedBids = (bids: BidDTO[]) => [...bids].sort((a, b) => a.amount - b.amount)
   228→  const bestBid = (bids: BidDTO[]) => (bids.length > 0 ? Math.min(...bids.map((b) => b.amount)) : null)
   229→
   230→  return (
   231→    <section aria-label="Площадка заявок">
   232→      <SectionHeader
   233→        eyebrow="РЕШЕНИЕ 1 · ПЛОЩАДКА ЗАЯВОК"
   234→        title="Агент выбирает заявку и делает ставку"
   235→        desc="Площадка закрытая: доступ только у проверенных перевозчиков с договором. Ставка, документы и статусы — в одном окне."
   236→      >
   237→        <div className="text-sm font-medium text-slate-700" aria-live="polite">
   238→          Открыто <span className="text-lg font-bold text-blue-700">{openCount}</span> заявок
   239→        </div>
   240→      </SectionHeader>
   241→
   242→      {!activeCarrierId && (
   243→        <div className="mb-4 rounded-xl border border-amber-200 bg-amber-50 px-4 py-3 text-sm text-amber-900" role="status">
   244→          Выберите перевозчика в шапке, чтобы действовать от его имени
   245→        </div>
   246→      )}
   247→
   248→      {loading && <LoadingBlock label="Загружаем заявки…" />}
   249→      {!loading && error && <ErrorBlock message={error} onRetry={() => void load()} />}
   250→
   251→      {!loading && !error && requests && (
   252→        <div className="space-y-4">
   253→          {/* Фильтры */}
   254→          <Card className="border-slate-200 shadow-sm">
   255→            <CardContent className="flex flex-wrap items-center gap-3 p-4">
   256→              <div className="relative min-w-[220px] flex-1">
   257→                <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-slate-400" aria-hidden />
   258→                <Input
   259→                  value={q}
   260→                  onChange={(e) => setQ(e.target.value)}
   261→                  placeholder="Поиск: номер, груз, маршрут…"
   262→                  className="pl-9"
   263→                  aria-label="Поиск по номеру, грузу или маршруту"
   264→                />
   265→              </div>
   266→              <Select value={mode} onValueChange={(v) => setMode(v as ModeFilter)}>
   267→                <SelectTrigger className="w-[200px]" aria-label="Фильтр по режиму заявки">
   268→                  <SelectValue placeholder="Все режимы" />
   269→                </SelectTrigger>
   270→                <SelectContent>
   271→                  {MODE_FILTERS.map((m) => (
   272→                    <SelectItem key={m.value} value={m.value}>{m.label}</SelectItem>
   273→                  ))}
   274→                </SelectContent>
   275→              </Select>
   276→              <Button variant="outline" onClick={() => void load()} disabled={refreshing} className="gap-2" aria-label="Обновить список заявок">
   277→                <RefreshCw className={`h-4 w-4 ${refreshing ? 'animate-spin' : ''}`} aria-hidden />
   278→                Обновить
   279→              </Button>
   280→              <span className="ml-auto hidden text-xs text-muted-foreground sm:block">автообновление каждые 15 секунд</span>
   281→            </CardContent>
   282→          </Card>
   283→
   284→          {/* Таблица заявок */}
   285→          <Card className="border-slate-200 shadow-sm">
   286→            <CardContent className="p-4 sm:p-6">
   287→              {filtered.length === 0 ? (
   288→                <EmptyState
   289→                  title="Заявок не найдено"
   290→                  hint="Измените параметры поиска или фильтр режима"
   291→                />
   292→              ) : (
   293→                <div className="-mx-2 overflow-x-auto px-2">
   294→                  <table className="w-full min-w-[980px] border-separate border-spacing-0 text-sm">
   295→                    <thead>
   296→                      <tr className="text-left text-xs uppercase tracking-wide text-slate-500">
   297→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">№ заявки</th>
   298→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Маршрут</th>
   299→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Груз</th>
   300→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Погрузка</th>
   301→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Тип ТС</th>
   302→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Режим</th>
   303→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">Ставка</th>
   304→                        <th scope="col" className="border-b border-slate-200 pb-2.5 pr-3 font-semibold">До окончания</th>
   305→                        <th scope="col" className="border-b border-slate-200 pb-2.5 font-semibold">Действие</th>
   306→                      </tr>
   307→                    </thead>
   308→                    <tbody>
   309→                      {filtered.map((r) => (
   310→                        <tr
   311→                          key={r.id}
   312→                          onClick={() => setDetailId(r.id)}
   313→                          className="cursor-pointer align-middle transition-colors hover:bg-blue-50/40"
   314→                        >
   315→                          <td className="border-b border-slate-100 py-3 pr-3">
   316→                            <span className="font-semibold text-blue-700">{r.number}</span>
   317→                          </td>
   318→                          <td className="border-b border-slate-100 py-3 pr-3">
   319→                            <span className="flex items-center gap-1.5 font-medium text-slate-800">
   320→                              {r.fromPoint}
   321→                              <ArrowRight className="h-3.5 w-3.5 shrink-0 text-slate-400" aria-hidden />
   322→                              {r.toPoint}
   323→                            </span>
   324→                          </td>
   325→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">
   326→                            {r.cargo}
   327→                            <span className="ml-1 whitespace-nowrap text-muted-foreground">{r.tonnage} т</span>
   328→                          </td>
   329→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">
   330→                            <div>{r.loadDate}</div>
   331→                            <div className="text-xs text-muted-foreground">{r.loadWindow}</div>
   332→                          </td>
   333→                          <td className="border-b border-slate-100 py-3 pr-3 text-slate-700">{r.truckType}</td>
   334→                          <td className="border-b border-slate-100 py-3 pr-3"><ModeBadge mode={r.mode} /></td>
   335→                          <td className="border-b border-slate-100 py-3 pr-3"><RateCell request={r} /></td>
   336→                          <td className="border-b border-slate-100 py-3 pr-3"><DeadlineCell request={r} now={now} /></td>
   337→                          <td className="border-b border-slate-100 py-3" onClick={(e) => e.stopPropagation()}>
   338→                            {r.status === 'PUBLISHED' && r.mode === 'FIXED' && (
   339→                              <Button
   340→                                size="sm"
   341→                                disabled={!activeCarrierId || busyId !== null}
   342→                                onClick={() => setTakeId(r.id)}
   343→                                aria-label={`Взять заявку ${r.number} по фиксированной ставке`}
   344→                              >
   345→                                Взять заявку
   346→                              </Button>
   347→                            )}
   348→                            {r.status === 'PUBLISHED' && r.mode === 'AUCTION' && (
   349→                              <Button
   350→                                size="sm"
   351→                                disabled={!activeCarrierId || busyId !== null}
   352→                                onClick={() => { setBidId(r.id); setBidAmount('') }}
   353→                                aria-label={`Сделать ставку по заявке ${r.number}`}
   354→                              >
   355→                                <Gavel className="h-4 w-4" aria-hidden />
   356→                                Сделать ставку
   357→                              </Button>
   358→                            )}
   359→                            {r.status === 'PUBLISHED' && r.mode === 'CLOSED' && (
   360→                              <Button
   361→                                size="sm"
   362→                                disabled={!activeCarrierId || busyId !== null}
   363→                                onClick={() => { setBidId(r.id); setBidAmount('') }}
   364→                                aria-label={`Подать закрытую ставку по заявке ${r.number}`}
   365→                              >
   366→                                <Lock className="h-4 w-4" aria-hidden />
   367→                                Подать закрытую ставку
   368→                              </Button>
   369→                            )}
   370→                            {r.status !== 'PUBLISHED' && <StatusBadge status={r.status} />}
   371→                          </td>
   372→                        </tr>
   373→                      ))}
   374→                    </tbody>
   375→                  </table>
   376→                </div>
   377→              )}
   378→              <p className="mt-3 text-xs text-muted-foreground">Нажмите на строку, чтобы увидеть детали заявки и историю ставок.</p>
   379→            </CardContent>
   380→          </Card>
   381→
   382→          {/* Подсказка про закрытую площадку */}
   383→          <div className="flex items-start gap-3 rounded-xl border border-blue-200 bg-blue-50 p-4 text-sm text-blue-900">
   384→            <ShieldCheck className="mt-0.5 h-5 w-5 shrink-0 text-blue-600" aria-hidden />
   385→            <p>
   386→              Площадка закрытая: доступ только у перевозчиков, которых мы заранее проверили и с которыми заключили договор.
   387→              Ставки видны логисту и участникам конкретной заявки — посторонние на площадку не попадают.
   388→            </p>
   389→          </div>
   390→        </div>
   391→      )}
   392→
   393→      {/* Подтверждение: взять заявку по фиксированной ставке */}
   394→      <AlertDialog open={takeReq !== null} onOpenChange={(o) => { if (!o) setTakeId(null) }}>
   395→        <AlertDialogContent>
   396→          <AlertDialogHeader>
   397→            <AlertDialogTitle>Взять заявку {takeReq?.number}?</AlertDialogTitle>
   398→            <AlertDialogDescription>
   399→              Заявка закрепляется за вами по фиксированной ставке {money(takeReq?.startRate ?? 0)}.
   400→              Дальше останется указать водителя и ТС и загрузить документы.
   401→            </AlertDialogDescription>
   402→          </AlertDialogHeader>
   403→          <AlertDialogFooter>
   404→            <AlertDialogCancel>Отмена</AlertDialogCancel>
   405→            <AlertDialogAction onClick={() => void doTake()} disabled={busyId !== null}>
   406→              {busyId === takeReq?.id ? 'Закрепляем…' : 'Взять заявку'}
   407→            </AlertDialogAction>
   408→          </AlertDialogFooter>
   409→        </AlertDialogContent>
   410→      </AlertDialog>
   411→
   412→      {/* Диалог ставки (торги / закрытые ставки) */}
   413→      <Dialog open={bidReq !== null} onOpenChange={(o) => { if (!o) closeBid() }}>
   414→        <DialogContent className="sm:max-w-md">
   415→          {bidReq && (
   416→            <>
   417→              <DialogHeader>
   418→                <DialogTitle>
   419→                  {bidReq.mode === 'CLOSED' ? 'Закрытая ставка' : 'Ставка'} · заявка {bidReq.number}
   420→                </DialogTitle>
   421→                <DialogDescription>
   422→                  {bidReq.fromPoint} → {bidReq.toPoint}, {bidReq.cargo}
   423→                </DialogDescription>
   424→              </DialogHeader>
   425→              <div className="space-y-4">
   426→                {bidReq.mode === 'AUCTION' && (
   427→                  <div className="rounded-lg border border-slate-200 bg-slate-50 px-3 py-2 text-sm text-slate-700">
   428→                    Текущая лучшая: <span className="font-semibold text-slate-900">{money(bidReq.bestRate ?? bidReq.startRate)}</span>
   429→                    {bidReq.bestRate == null && <span className="ml-1 text-muted-foreground">— ставок ещё нет</span>}
   430→                  </div>
   431→                )}
   432→                {bidReq.mode === 'CLOSED' && (
   433→                  <div className="flex items-start gap-2 rounded-lg border border-slate-200 bg-slate-50 px-3 py-2 text-sm text-slate-700">
   434→                    <Lock className="mt-0.5 h-4 w-4 shrink-0 text-slate-400" aria-hidden />
   435→                    Участники не видят цены друг друга — защита от сговора. Победителя определит логист по итогам.
   436→                  </div>
   437→                )}
   438→                <div className="space-y-1.5">
   439→                  <Label htmlFor="bid-amount">Ваша ставка, ₽</Label>
   440→                  <Input
   441→                    id="bid-amount"
   442→                    type="number"
   443→                    min={1}
   444→                    step={100}
   445→                    inputMode="numeric"
   446→                    placeholder="Например, 85000"
   447→                    value={bidAmount}
   448→                    onChange={(e) => setBidAmount(e.target.value)}
   449→                    onKeyDown={(e) => { if (e.key === 'Enter') void doBid() }}
   450→                  />
   451→                  <p className="text-xs text-muted-foreground">
   452→                    {bidReq.mode === 'CLOSED' ? 'Ставку увидит только логист при подведении итогов' : 'Должна быть ниже лучшей'}
   453→                  </p>
   454→                </div>
   455→              </div>
   456→              <DialogFooter>
   457→                <Button variant="outline" onClick={closeBid}>Отмена</Button>
   458→                <Button onClick={() => void doBid()} disabled={busyId !== null || !bidAmount.trim()}>
   459→                  {busyId === bidReq.id ? 'Отправляем…' : 'Подтвердить ставку'}
   460→                </Button>
   461→              </DialogFooter>
   462→            </>
   463→          )}
   464→        </DialogContent>
   465→      </Dialog>
   466→
   467→      {/* Детали заявки */}
   468→      <Dialog open={detailReq !== null} onOpenChange={(o) => { if (!o) setDetailId(null) }}>
   469→        <DialogContent className={`max-h-[90vh] overflow-y-auto sm:max-w-2xl ${THIN_SCROLL}`}>
   470→          {detailReq && (
   471→            <>
   472→              <DialogHeader>
   473→                <DialogTitle className="flex flex-wrap items-center gap-2">
   474→                  Заявка <span className="text-blue-700">{detailReq.number}</span>
   475→                  <StatusBadge status={detailReq.status} />
   476→                  <ModeBadge mode={detailReq.mode} />
   477→                </DialogTitle>
   478→                <DialogDescription className="flex items-center gap-1.5">
   479→                  {detailReq.fromPoint} <ArrowRight className="h-3.5 w-3.5" aria-hidden /> {detailReq.toPoint}
   480→                </DialogDescription>
   481→              </DialogHeader>
   482→
   483→              <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   484→                {[
   485→                  { label: 'Груз', value: `${detailReq.cargo}, ${detailReq.tonnage} т` },
   486→                  { label: 'Тип ТС', value: detailReq.truckType },
   487→                  { label: 'Погрузка', value: `${detailReq.loadDate}, ${detailReq.loadWindow}` },
   488→                  { label: 'Стартовая ставка', value: money(detailReq.startRate) },
   489→                  { label: 'Текущая лучшая', value: detailReq.bestRate != null ? money(detailReq.bestRate) : '—' },
   490→                  { label: detailReq.winnerName ? 'Перевозчик' : 'Окончание ставок', value: detailReq.winnerName ?? '—' },
   491→                ].map((f) => (
   492→                  <div key={f.label} className="rounded-lg border border-slate-200 bg-slate-50/60 p-2.5">
   493→                    <div className="text-[11px] uppercase tracking-wide text-muted-foreground">{f.label}</div>
   494→                    <div className="mt-0.5 text-sm font-semibold text-slate-900">{f.value}</div>
   495→                  </div>
   496→                ))}
   497→              </div>
   498→
   499→              <div className="rounded-lg border border-blue-200 bg-blue-50 p-3">
   500→                <div className="flex items-center gap-1.5 text-sm font-semibold text-blue-900">
   501→                  <ShieldCheck className="h-4 w-4" aria-hidden />
   502→                  Требования к документам
   503→                </div>
   504→                <p className="mt-1 text-sm text-blue-800">
   505→                  Пропуск в порт: паспорт, ВУ, СТС. Нейросеть сверит фото с данными водителя и автомобиля —
   506→                  комплект загружается в эту же заявку.
   507→                </p>
   508→              </div>
   509→
   510→              {detailReq.mode === 'AUCTION' && (
   511→                <div>
   512→                  <h4 className="mb-2 flex items-center gap-1.5 text-sm font-semibold text-slate-900">
   513→                    <Gavel className="h-4 w-4 text-blue-600" aria-hidden />
   514→                    Ставки ({detailReq.bids.length})
   515→                  </h4>
   516→                  {detailReq.bids.length === 0 ? (
   517→                    <p className="rounded-lg border border-dashed border-slate-300 px-3 py-4 text-center text-sm text-muted-foreground">
   518→                      Ставок пока нет — будьте первым.
   519→                    </p>
   520→                  ) : (
   521→                    <div className="overflow-x-auto rounded-lg border border-slate-200">
   522→                      <Table>
   523→                        <TableHeader>
   524→                          <TableRow>
   525→                            <TableHead>Перевозчик</TableHead>
   526→                            <TableHead>Ставка</TableHead>
   527→                            <TableHead>Время</TableHead>
   528→                          </TableRow>
   529→                        </TableHeader>
   530→                        <TableBody>
   531→                          {sortedBids(detailReq.bids).map((b) => {
   532→                            const isBest = bestBid(detailReq.bids) === b.amount
   533→                            return (
   534→                              <TableRow key={b.id} className={isBest ? 'bg-emerald-50' : undefined}>
   535→                                <TableCell className="font-medium">
   536→                                  <span className="flex items-center gap-2">
   537→                                    {b.carrierName}
   538→                                    {isBest && <Badge className="border-emerald-200 bg-emerald-100 text-[10px] text-emerald-800" variant="outline">лучшая</Badge>}
   539→                                  </span>
   540→                                </TableCell>
   541→                                <TableCell className={isBest ? 'font-semibold text-emerald-700' : 'text-slate-700'}>
   542→                                  {money(b.amount)}
   543→                                </TableCell>
   544→                                <TableCell className="text-muted-foreground">{fmtDateTime(b.createdAt)}</TableCell>
   545→                              </TableRow>
   546→                            )
   547→                          })}
   548→                        </TableBody>
   549→                      </Table>
   550→                    </div>
   551→                  )}
   552→                </div>
   553→              )}
   554→
   555→              {detailReq.mode === 'CLOSED' && (
   556→                <div className="flex items-center gap-2 rounded-lg border border-slate-200 bg-slate-50 p-3 text-sm text-slate-600">
   557→                  <Lock className="h-4 w-4 shrink-0" aria-hidden />
   558→                  Ставки скрыты до подведения итогов — участники не видят цены друг друга.
   559→                </div>
   560→              )}
   561→
   562→              {detailReq.mode === 'FIXED' && (
   563→                <div className="rounded-lg border border-emerald-200 bg-emerald-50 p-3 text-sm text-emerald-900">
   564→                  Фиксированная ставка: <span className="font-semibold">{money(detailReq.startRate)}</span> — торги не проводятся.
   565→                </div>
   566→              )}
   567→
   568→              {detailReq.status !== 'PUBLISHED' && detailReq.mode === 'AUCTION' && detailReq.bids.length > 0 && (
   569→                <p className="text-xs text-muted-foreground">
   570→                  Торги завершены — победитель согласован логистом.
   571→                </p>
   572→              )}
   573→            </>
   574→          )}
   575→        </DialogContent>
   576→      </Dialog>
   577→    </section>
   578→  )
