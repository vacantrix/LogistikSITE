    85→/** Текст-подсказка для агента по статусу заявки */
    86→function agentLine(r: RequestDTO): { text: string; cls: string } {
    87→  switch (r.status) {
    88→    case 'AWARDED':
    89→      return { text: 'Укажите водителя и ТС, загрузите документы', cls: 'border-amber-200 bg-amber-50 text-amber-900' }
    90→    case 'DOCS_REVIEW': {
    91→      const bad = r.documents.some((d) => d.status === 'REUPLOAD' || d.status === 'REJECTED')
    92→      return bad
    93→        ? { text: 'Требует исправления: причина по каждому документу, догрузка — в ту же заявку', cls: 'border-amber-300 bg-amber-50 text-amber-900' }
    94→        : { text: 'Нейросеть проверяет документы', cls: 'border-blue-200 bg-blue-50 text-blue-900' }
    95→    }
    96→    case 'AT_LOGIST':
    97→      return { text: 'Комплект у логиста — идёт сверка фото с полями', cls: 'border-blue-200 bg-blue-50 text-blue-900' }
    98→    case 'CONFIRMED':
    99→      return { text: 'Принята — данные переданы в 1С', cls: 'border-emerald-200 bg-emerald-50 text-emerald-900' }
   100→    default:
   101→      return { text: 'Рейс закрыт', cls: 'border-slate-300 bg-slate-100 text-slate-600' }
   102→  }
   103→}

   105→/** Цвет процента уверенности распознавания */
   106→function confClass(c: number): string {
   107→  if (c >= 0.95) return 'text-emerald-600'
   108→  if (c >= 0.6) return 'text-blue-600'
   109→  return 'text-amber-600'
   110→}

   112→/** Цвет списка замечаний по статусу документа */
   113→function issueClass(status: DocStatus): string {
   114→  if (status === 'REJECTED') return 'text-red-600'
   115→  if (status === 'REUPLOAD') return 'text-amber-600'
   116→  if (status === 'MANUAL') return 'text-blue-600'
   117→  return 'text-slate-500'
   118→}
