   136→   136→function DirFilterBar({ query, onQuery, expiry, onExpiry, withExpiry, count, children, extraActive, onReset }: {
   137→   137→  query: string
   138→   138→  onQuery: (v: string) => void
   139→   139→  expiry: 'all' | 'soon' | 'expired'
   140→   140→  onExpiry: (v: 'all' | 'soon' | 'expired') => void
   141→   141→  withExpiry: boolean
   142→   142→  count?: string
   143→   143→  /** дополнительные фильтры вкладки (категория груза, доступ перевозчика) */
   144→   144→  children?: ReactNode
   145→   145→  extraActive?: boolean
   146→   146→  onReset?: () => void
   147→   147→}) {
   148→   148→  const active = query.trim() !== '' || expiry !== 'all' || !!extraActive
   149→   149→  const reset = onReset ?? (() => { onQuery(''); onExpiry('all') })
   150→   150→  return (
   151→   151→    <div className="mb-3 rounded-xl border border-slate-200 bg-white p-3 shadow-sm">
   152→   152→      <div className="flex flex-wrap items-center gap-2">
   153→   153→        <div className="relative min-w-[220px] flex-1 sm:max-w-sm">
   154→   154→          <Search className="pointer-events-none absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-slate-400" aria-hidden />
   155→   155→          <Input
   156→   156→            value={query}
   157→   157→            onChange={(e) => onQuery(e.target.value)}
   158→   158→            placeholder="Поиск по справочнику…"
   159→   159→            className="h-9 pl-8"
   160→   160→            aria-label="Поиск по справочнику"
   161→   161→          />
   162→   162→        </div>
   163→   163→        {withExpiry && (
   164→   164→          <Select value={expiry} onValueChange={(v) => onExpiry(v as 'all' | 'soon' | 'expired')}>
   165→   165→            <SelectTrigger className="h-9 w-[200px]" aria-label="Фильтр по сроку документов">
   166→   166→              <SelectValue />
   167→   167→            </SelectTrigger>
   168→   168→            <SelectContent>
   169→   169→              <SelectItem value="all">Все сроки документов</SelectItem>
   170→   170→              <SelectItem value="soon">Истекает (90 дней)</SelectItem>
   171→   171→              <SelectItem value="expired">Истёкшие документы</SelectItem>
   172→   172→            </SelectContent>
   173→   173→          </Select>
   174→   174→        )}
   175→   175→        {children}
   176→   176→        {active && (
   177→   177→          <Button size="sm" variant="ghost" className="h-9" onClick={reset}>
   178→   178→            Сбросить
   179→   179→          </Button>
   180→   180→        )}
   181→   181→        {count && <div className="ml-auto text-xs text-slate-500">{count}</div>}
   182→   182→      </div>
   183→   183→    </div>
   184→   184→  )
   185→   185→}

   187→   187→export default function DirectoriesTab({ activeCarrierId, role, onNavigate }: TabProps) {
   188→   188→  const { toast } = useToast()
   189→   189→  const [carriers, setCarriers] = useState<CarrierDTO[] | null>(null)
   190→   190→  const [error, setError] = useState<string | null>(null)
   191→   191→  const [tab, setTab] = useState<TabKind>('carriers')
   192→   192→  const [addKind, setAddKind] = useState<Kind | null>(null)
   193→   193→  const [editing, setEditing] = useState<{ kind: Kind; id: string } | null>(null)
   194→   194→  const [form, setForm] = useState<Record<string, string>>({})
   195→   195→  const [saving, setSaving] = useState(false)
   196→   196→  const isLogist = role === 'logist'
   197→   197→  // ——— фильтры справочника: текстовый поиск + срок документов + фильтры вкладок ———
   198→   198→  const [dirQuery, setDirQuery] = useState('')
   199→   199→  const [dirExpiry, setDirExpiry] = useState<'all' | 'soon' | 'expired'>('all')
   200→   200→  const [dirAccess, setDirAccess] = useState<'all' | 'open' | 'closed'>('all')
   201→   201→
   202→   202→  // перевозчику доступен только его собственный парк — данные конкурентов скрыты
   203→   203→  const visibleCarriers = useMemo(
   204→   204→    () => (isLogist ? (carriers ?? []) : (carriers ?? []).filter((c) => c.id === activeCarrierId)),
   205→   205→    [carriers, isLogist, activeCarrierId],
   206→   206→  )
   207→   207→
   208→   208→  const load = useCallback(async () => {
   209→   209→    try {
   210→   210→      const dir = await api.directory()
   211→   211→      setCarriers(dir.carriers)
   212→   212→      setError(null)
   213→   213→    } catch (e) {
   214→   214→      setError(e instanceof Error ? e.message : 'Неизвестная ошибка')
   215→   215→    }
   216→   216→  }, [])
   217→   217→
   218→   218→  useEffect(() => { void load() }, [load])
   219→   219→
   220→   220→  // —— фильтрация справочника ——
   221→   221→  const dirQ = dirQuery.trim().toLowerCase()
   222→   222→  const dirByQuery = useCallback(
   223→   223→    (...fields: (string | null | undefined)[]) =>
   224→   224→      !dirQ || fields.some((f) => (f ?? '').toLowerCase().includes(dirQ)),
   225→   225→    [dirQ],
   226→   226→  )
   227→   227→  const dirByExpiry = useCallback(
   228→   228→    (dates: (string | null | undefined)[]) => {
   229→   229→      if (dirExpiry === 'all') return true
   230→   230→      const worst = dates.reduce<ExpiryState>((acc, d) => worseState(acc, expiryInfo(d).state), 'none')
   231→   231→      return dirExpiry === 'expired' ? worst === 'expired' : worst === 'soon' || worst === 'warn'
   232→   232→    },
   233→   233→    [dirExpiry],
   234→   234→  )
   235→   235→
   236→   236→  const carriersFiltered = useMemo(
   237→   237→    () => (carriers ?? [])
   238→   238→      .filter((c) => dirByQuery(c.name, c.inn, c.contractNo))
   239→   239→      .filter((c) => dirAccess === 'all' || (dirAccess === 'open' ? c.access : !c.access)),
   240→   240→    [carriers, dirByQuery, dirAccess],
   241→   241→  )
   242→   242→  // —— полный (нефильтрованный) парк видимой области: сводка и счётчики ——
   243→   243→  const allDrivers = useMemo(
   244→   244→    () => visibleCarriers.flatMap((c) => c.drivers.map((d) => ({ d, carrierName: c.name }))),
   245→   245→    [visibleCarriers],
   246→   246→  )
   247→   247→  const allTrucks = useMemo(
   248→   248→    () => visibleCarriers.flatMap((c) => c.trucks.map((t) => ({ t, carrierName: c.name }))),
   249→   249→    [visibleCarriers],
   250→   250→  )
   251→   251→  const allTrailers = useMemo(
   252→   252→    () => visibleCarriers.flatMap((c) => c.trailers.map((z) => ({ z, carrierName: c.name }))),
   253→   253→    [visibleCarriers],
   254→   254→  )
   255→   255→
   256→   256→  // —— отфильтрованные выборки для таблиц ——
   257→   257→  const driversFlat = useMemo(
   258→   258→    () => allDrivers
   259→   259→      .filter(({ d }) => dirByQuery(d.fullName, d.passport, d.license, d.phone) && dirByExpiry([d.licenseTill, d.medTill])),
   260→   260→    [allDrivers, dirByQuery, dirByExpiry],
   261→   261→  )
   262→   262→  const driversTotal = allDrivers.length
   263→   263→  const trucksFlat = useMemo(
   264→   264→    () => allTrucks
   265→   265→      .filter(({ t }) => dirByQuery(t.brand, t.plate, t.vin, t.owner) && dirByExpiry([t.techTill])),
   266→   266→    [allTrucks, dirByQuery, dirByExpiry],
   267→   267→  )
   268→   268→  const trucksTotal = allTrucks.length
   269→   269→  const trailersFlat = useMemo(
   270→   270→    () => allTrailers
   271→   271→      .filter(({ z }) => dirByQuery(z.plate, z.type, z.owner) && dirByExpiry([z.techTill])),
   272→   272→    [allTrailers, dirByQuery, dirByExpiry],
   273→   273→  )
   274→   274→  const trailersTotal = allTrailers.length
   275→   275→
   276→   276→  /** Краткая сводка: кто и когда добавлен агентом — последние записи по времени */
   277→   277→  const recentAdditions = useMemo<RecentAddition[]>(() => {
   278→   278→    const items: RecentAddition[] = [
   279→   279→      ...allDrivers.map(({ d, carrierName }) => ({ key: `d-${d.id}`, tab: 'drivers' as const, icon: User, title: d.fullName, kindLabel: 'Водитель', carrierName, createdAt: d.createdAt })),
   280→   280→      ...allTrucks.map(({ t, carrierName }) => ({ key: `t-${t.id}`, tab: 'trucks' as const, icon: Truck, title: `${t.brand} · ${t.plate}`, kindLabel: 'Тягач', carrierName, createdAt: t.createdAt })),
   281→   281→      ...allTrailers.map(({ z, carrierName }) => ({ key: `z-${z.id}`, tab: 'trailers' as const, icon: Wrench, title: `${z.plate} · ${z.type}`, kindLabel: 'Прицеп', carrierName, createdAt: z.createdAt })),
   282→   282→    ]
   283→   283→    return items.sort((a, b) => b.createdAt.localeCompare(a.createdAt)).slice(0, 6)
   284→   284→  }, [allDrivers, allTrucks, allTrailers])
   285→   285→
   286→   286→  /** Что и когда истекает: все документы с плохими/близкими сроками */
   287→   287→  const expiryItems = useMemo<ExpiryItem[]>(() => {
   288→   288→    const items: ExpiryItem[] = []
   289→   289→    const bad = (i: ReturnType<typeof expiryInfo>) => i.state === 'expired' || i.state === 'warn'
   290→   290→    for (const c of visibleCarriers) {
   291→   291→      for (const d of c.drivers) {
   292→   292→        const v = expiryInfo(d.licenseTill)
   293→   293→        if (bad(v)) items.push({ key: `${d.id}-lic`, tab: 'drivers', icon: User, title: d.fullName, docLabel: 'Водительское удостоверение', info: v, carrierName: c.name })
   294→   294→        const m = expiryInfo(d.medTill)
   295→   295→        if (bad(m)) items.push({ key: `${d.id}-med`, tab: 'drivers', icon: User, title: d.fullName, docLabel: 'Медсправка', info: m, carrierName: c.name })
   296→   296→      }
   297→   297→      for (const t of c.trucks) {
   298→   298→        const v = expiryInfo(t.techTill)
   299→   299→        if (bad(v)) items.push({ key: t.id, tab: 'trucks', icon: Truck, title: `${t.brand} · ${t.plate}`, docLabel: 'Техосмотр', info: v, carrierName: c.name })
   300→   300→      }
   301→   301→      for (const z of c.trailers) {
   302→   302→        const v = expiryInfo(z.techTill)
   303→   303→        if (bad(v)) items.push({ key: z.id, tab: 'trailers', icon: Wrench, title: `${z.type} · ${z.plate}`, docLabel: 'Техосмотр', info: v, carrierName: c.name })
   304→   304→      }
   305→   305→    }
   306→   306→    return items.sort((a, b) => (a.info.days ?? 0) - (b.info.days ?? 0))
   307→   307→  }, [visibleCarriers])
   308→   308→
   309→   309→  const expiredCount = expiryItems.filter((i) => i.info.state === 'expired').length
   310→   310→
   311→   311→  const openAdd = (kind: Kind) => {
   312→   312→    setForm({ carrierId: isLogist ? (carriers?.[0]?.id ?? '') : (activeCarrierId ?? '') })
   313→   313→    setEditing(null)
   314→   314→    setAddKind(kind)
   315→   315→  }
   316→   316→
   317→   317→  const openEdit = (kind: Kind, id: string, record: Record<string, string>) => {
   318→   318→    setForm({ ...record })
   319→   319→    setAddKind(null)
   320→   320→    setEditing({ kind, id })
   321→   321→  }
   322→   322→
   323→   323→  const closeDialog = () => {
   324→   324→    setAddKind(null)
   325→   325→    setEditing(null)
   326→   326→  }
   327→   327→
   328→   328→  const submitAdd = async () => {
   329→   329→    if (!addKind) return
   330→   330→    setSaving(true)
   331→   331→    try {
   332→   332→      await api.createDirectory(addKind, form)
   333→   333→      toast({
   334→   334→        title: 'Запись добавлена в справочник',
   335→   335→        description: `${KIND_TITLES[addKind]}: запись создана и доступна при назначении экипажа`,
   336→   336→      })
   337→   337→      closeDialog()
   338→   338→      await load()
   339→   339→    } catch (e) {
   340→   340→      toast({
   341→   341→        title: 'Не удалось добавить запись',
   342→   342→        description: e instanceof Error ? e.message : 'Проверьте заполнение полей',
   343→   343→        variant: 'destructive',
   344→   344→      })
   345→   345→    } finally {
   346→   346→      setSaving(false)
   347→   347→    }
   348→   348→  }
   349→   349→
   350→   350→  const submitEdit = async () => {
   351→   351→    if (!editing) return
   352→   352→    setSaving(true)
   353→   353→    try {
   354→   354→      await api.updateDirectory(editing.kind, editing.id, form)
   355→   355→      toast({
   356→   356→        title: 'Запись обновлена',
   357→   357→        description: `${KIND_TITLES[editing.kind]}: данные и сроки документов сохранены`,
   358→   358→      })
   359→   359→      closeDialog()
   360→   360→      await load()
   361→   361→    } catch (e) {
   362→   362→      toast({
   363→   363→        title: 'Не удалось обновить запись',
   364→   364→        description: e instanceof Error ? e.message : 'Проверьте заполнение полей',
   365→   365→        variant: 'destructive',
   366→   366→      })
   367→   367→    } finally {
   368→   368→      setSaving(false)
   369→   369→    }
   370→   370→  }
   371→   371→
   372→   372→  if (carriers === null && !error) return <LoadingBlock label="Загрузка кадров и техники…" />
   373→   373→  if (error) return <ErrorBlock message={error} onRetry={() => { setError(null); void load() }} />
