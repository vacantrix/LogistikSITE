    70→    70→/** Цвет рамки слота по вердикту нейросети */
    71→    71→function slotBorder(s: SlotState): string {
    72→    72→  if (s.checking) return 'border-blue-300 bg-blue-50/30'
    73→    73→  if (s.error) return 'border-red-300 bg-red-50/40'
    74→    74→  if (!s.verdict) return 'border-slate-200'
    75→    75→  switch (s.verdict.status) {
    76→    76→    case 'PASSED': return 'border-emerald-300 bg-emerald-50/40'
    77→    77→    case 'MANUAL': return 'border-blue-300 bg-blue-50/30'
    78→    78→    case 'REUPLOAD': return 'border-amber-300 bg-amber-50/40'
    79→    79→    case 'REJECTED': return 'border-red-300 bg-red-50/40'
    80→    80→    default: return 'border-slate-200'
    81→    81→  }
    82→    82→}

    91→    91→function SlotVerdict({ s, type }: { s: SlotState; type: DocType }) {
    92→    92→  if (s.checking) {
    93→    93→    return (

   104→   104→        {s.error}
   105→   105→      </div>
   106→   106→    )
   107→   107→  }

   109→   109→  if (!v) {
   110→   110→    return <p className="mt-2 text-[11px] leading-snug text-slate-400">{SLOT_META[type].hint}. Нейросеть проверит документ сразу после загрузки.</p>

   112→   112→
   113→   113→  if (v.status === 'REJECTED') {
   114→   114→    return (
   115→   115→      <div className="mt-2 space-y-1.5 rounded-lg border border-red-200 bg-red-50 px-2.5 py-2">
   116→   116→        <div className="flex items-center gap-1.5 text-xs font-semibold text-red-800">
   117→   117→          <XCircle className="h-3.5 w-3.5" aria-hidden />
   118→   118→          Отказ нейросети — замените файл
   119→   119→        </div>
   120→   120→        <p className="text-[11px] leading-snug text-red-700">{v.message}</p>
   121→   121→        <p className="text-[11px] leading-snug text-red-600">Заявка не будет отправлена, пока документ не заменён.</p>
   122→   122→      </div>
   123→   123→    )
   124→   124→  }
   125→   125→  if (v.status === 'REUPLOAD') {
   126→   126→    return (
   127→   127→      <div className="mt-2 space-y-1 rounded-lg border border-amber-200 bg-amber-50 px-2.5 py-2">
   128→   128→        <div className="flex items-center gap-1.5 text-xs font-semibold text-amber-800">
   129→   129→          <AlertTriangle className="h-3.5 w-3.5" aria-hidden />
   130→   130→          <DocStatusBadge status={v.status} />
   131→   131→        </div>
   132→   132→        <p className="text-[11px] leading-snug text-amber-700">{v.message}</p>
   133→   133→      </div>
   134→   134→    )
   135→   135→  }
   136→   136→  if (v.status === 'MANUAL') {
   137→   137→    return (
   138→   138→      <div className="mt-2 space-y-1 rounded-lg border border-blue-200 bg-blue-50 px-2.5 py-2">
   139→   139→        <div className="flex items-center gap-1.5 text-xs font-semibold text-blue-800">
   140→   140→          <Info className="h-3.5 w-3.5" aria-hidden />
   141→   141→          <DocStatusBadge status={v.status} />
   142→   142→        </div>
   143→   143→        <p className="text-[11px] leading-snug text-blue-700">{v.message}</p>
   144→   144→      </div>
   145→   145→    )
   146→   146→  }
   147→   147→  // PASSED: показываем распознанные поля с уверенностью
   148→   148→  const fieldRows = Object.entries(v.fields)
   149→   149→    .filter(([, val]) => val)
   150→   150→    .slice(0, 3)
   151→   151→  return (
   152→   152→    <div className="mt-2 space-y-1 rounded-lg border border-emerald-200 bg-emerald-50 px-2.5 py-2">
   153→   153→      <div className="flex items-center gap-1.5 text-xs font-semibold text-emerald-800">
   154→   154→        <CheckCircle2 className="h-3.5 w-3.5" aria-hidden />
   155→   155→        Вид документа подтверждён · качество {v.quality}
   156→   156→      </div>
   157→   157→      {fieldRows.map(([label, val]) => (
   158→   158→        <div key={label} className="flex items-baseline justify-between gap-2 text-[11px]">
   159→   159→          <span className="text-slate-500">{label}:</span>
   160→   160→          <span className="min-w-0 truncate font-medium text-slate-700">{val}</span>
   161→   161→          <span className={`shrink-0 font-semibold ${confCls(v.confidence[label] ?? 0)}`}>
   162→   162→            {v.confidence[label] != null ? `${Math.round((v.confidence[label] ?? 0) * 100)}%` : ''}
   163→   163→          </span>
   164→   164→        </div>
   165→   165→      ))}
   166→   166→    </div>
   167→   167→  )
   168→   168→}
