    65→    65→function RateCell({ request }: { request: RequestDTO }) {

    68→    68→  }

    70→    70→    return (
    71→    71→      <div>
    72→    72→        <div className="flex items-center gap-1 font-semibold text-slate-900">
    73→    73→          <Lock className="h-3.5 w-3.5 text-slate-400" aria-hidden />

    77→    77→      </div>
    78→    78→    )
    79→    79→  }
    80→    80→  if (request.bestRate == null) {
    81→    81→    return (
    82→    82→      <div>
    83→    83→        <div className="text-sm font-medium text-slate-500">Ставок нет</div>

    89→    89→    <div>

    92→    92→    </div>

    94→    94→}

    97→    97→function DeadlineCell({ request, now }: { request: RequestDTO; now: number | null }) {
    98→    98→  if (request.mode === 'FIXED' || !request.deadline) return <span className="text-slate-400">—</span>
    99→    99→  if (now === null) return <span className="text-slate-400">—</span>
   100→   100→  const tl = timeLeft(request.deadline)
   101→   101→  if (!tl) return <span className="text-slate-400">—</span>
   102→   102→  return (
   103→   103→    <span className={`tabular-nums ${tl.urgent ? 'font-semibold text-red-600' : 'text-slate-700'}`}>
   104→   104→      {tl.text}
   105→   105→    </span>
   106→   106→  )
   107→   107→}

   243→   243→      toast({ title: 'Ошибка', description: 'Заполните направление, груз и стартовую ставку', variant: 'destructive' })

   249→   249→        fromPoint: form.fromPoint.trim(),
   250→   250→        toPoint: form.toPoint.trim(),

   267→   267→      setCreating(false)
   268→   268→    }
   269→   269→  }
   270→   270→
   271→   271→  return (
