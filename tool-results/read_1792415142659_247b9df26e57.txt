   104→/** Цвет процента уверенности распознавания */
   105→function confCls(c: number): string {
   106→  if (c >= 0.95) return 'text-emerald-600'
   107→  if (c >= 0.6) return 'text-blue-600'
   108→  return 'text-amber-600'
   109→}
   110→
   111→function ConfChip({ c }: { c: number }) {
   112→  if (!c) return <span className="text-[10px] font-semibold text-slate-400">не распознано</span>
   113→  return <span className={`text-[10px] font-semibold ${confCls(c)}`}>{Math.round(c * 100)}%</span>
   114→}

   165→export interface DriverIntakeDialogProps {

   168→  carriers: CarrierDTO[]

   177→
   178→export default function DriverIntakeDialog({ open, onOpenChange, carriers, defaultCarrierId, lockCarrier, request, onSaved }: DriverIntakeDialogProps) {
   179→  const { toast } = useToast()
   180→  const [step, setStep] = useState<'docs' | 'scan' | 'review'>('docs')
   181→  const [carrierId, setCarrierId] = useState('')
   182→  const [uploads, setUploads] = useState<Uploads>({})
   183→  const [scanStage, setScanStage] = useState(SCAN_STAGES[0])
   184→  const [result, setResult] = useState<IntakeResult | null>(null)
   185→  const [draftDriver, setDraftDriver] = useState<IntakeDriverDraft | null>(null)
   186→  const [draftTruck, setDraftTruck] = useState<IntakeTruckDraft | null>(null)
   187→  const [attach, setAttach] = useState(true)
   188→  const [saving, setSaving] = useState(false)
   189→  const fileInputs = useRef<Partial<Record<DocType, HTMLInputElement | null>>>({})
   190→
   191→  const canRecognize = !!(carrierId && uploads.PASSPORT_PHOTO && uploads.LICENSE)
   192→
   193→  // сброс и инициализация при открытии
   194→  useEffect(() => {
   195→    if (!open) return
   196→    setStep('docs')
   197→    setUploads({})
   198→    setResult(null)
   199→    setDraftDriver(null)
   200→    setDraftTruck(null)
   201→    setAttach(true)
   202→    setSaving(false)
   203→    setCarrierId(defaultCarrierId ?? carriers[0]?.id ?? '')
   204→  }, [open, defaultCarrierId, carriers])
   205→
   206→  // анимация этапов сканирования
   207→  useEffect(() => {
   208→    if (step !== 'scan') return
   209→    let i = 0
   210→    setScanStage(SCAN_STAGES[0])
   211→    const iv = setInterval(() => {
   212→      i = (i + 1) % SCAN_STAGES.length
   213→      setScanStage(SCAN_STAGES[i])
   214→    }, 700)
   215→    return () => clearInterval(iv)
   216→  }, [step])
   217→
   218→  const setUpload = useCallback((type: DocType, up: Upload | null) => {
   219→    setUploads((prev) => {
   220→      const next = { ...prev }
   221→      if (up) next[type] = up
   222→      else delete next[type]
   223→      return next
   224→    })
   225→  }, [])
   226→
   227→  const handleFile = async (type: DocType, file: File) => {
   228→    try {
   229→      const dataUri = await fileToCompressedDataUri(file)
   230→      setUpload(type, { fileName: file.name, dataUri })
   231→    } catch (e) {
   232→      toast({ title: 'Ошибка', description: e instanceof Error ? e.message : 'Не удалось прочитать файл', variant: 'destructive' })
   233→    }
   234→  }
   235→
   236→  const fillDemo = () => {
   237→    setUploads(Object.fromEntries(
   238→      (Object.keys(DEMO) as DocType[]).map((t) => [t, { fileName: DEMO[t].fileName, dataUri: sampleSvg(DEMO[t].title, DEMO[t].lines, DEMO[t].accent) }]),
   239→    ))
   240→    toast({ title: 'Демо-образцы загружены', description: 'Паспорт, ВУ и СТС водителя Петрова П.С. — нажмите «Распознать документы»' })
   241→  }
   242→
   243→  const recognize = async () => {
   244→    if (!canRecognize) return
   245→    setStep('scan')
   246→    try {
   247→      const [res] = await Promise.all([
   248→        api.intakeDriver(Object.fromEntries(Object.entries(uploads).filter(([, v]) => v)) as Partial<Record<string, Upload>>),
   249→        sleep(2200), // минимальное время «работы» нейросети для наглядности
   250→      ])
   251→      setResult(res)
   252→      setDraftDriver(res.driver)
   253→      setDraftTruck(res.truck)
   254→      setStep('review')
   255→      const bad = res.documents.filter((d) => d.status === 'REUPLOAD' || d.status === 'REJECTED')
   256→      if (bad.length > 0) {
   257→        toast({
   258→          title: 'Часть документов не прошла проверку',
   259→          description: `${bad.map((d) => DOC_TYPE_LABEL[d.type]).join(', ')} — замените фото или введите данные вручную`,
   260→          variant: 'destructive',
   261→        })
   262→      } else {
   263→        toast({ title: 'Нейросеть заполнила карточку водителя', description: 'Проверьте распознанные поля и подтвердите' })
   264→      }
   265→    } catch (e) {
   266→      setStep('docs')
   267→      toast({ title: 'Ошибка распознавания', description: e instanceof Error ? e.message : 'Неизвестная ошибка', variant: 'destructive' })
   268→    }
   269→  }
   270→
   271→  const save = async () => {
   272→    if (!draftDriver || !carrierId) return
   273→    setSaving(true)
   274→    try {
   275→      const driverRes = await api.createDirectory('driver', {
   276→        carrierId,
   277→        fullName: draftDriver.fullName,
   278→        birthDate: draftDriver.birthDate,
   279→        passport: draftDriver.passport,
   280→        license: draftDriver.license,
   281→        licenseTill: draftDriver.licenseTill,
   282→        categories: draftDriver.categories,
   283→      })
   284→      const driverId = driverRes.driver?.id
   285→      if (!driverId) throw new Error('Не удалось сохранить водителя')
   286→
   287→      let truckId: string | undefined
   288→      if (draftTruck?.plate) {
   289→        const truckRes = await api.createDirectory('truck', {
   290→          carrierId,
   291→          brand: draftTruck.brand,
   292→          plate: draftTruck.plate,
   293→          vin: draftTruck.vin,
   294→          sts: draftTruck.sts,
   295→          owner: draftTruck.owner,
   296→        })
   297→        truckId = truckRes.truck?.id
   298→      }
   299→
   300→      let savedRequest: RequestDTO | undefined
   301→      const attachToRequest = !!(request && attach)
   302→      if (attachToRequest) {
   303→        const a = await api.action(request!.id, 'assign', { driverId, truckId })
   304→        savedRequest = a.request
   305→        for (const [type, up] of Object.entries(uploads)) {
   306→          if (!up) continue
   307→          const r = await api.uploadDocument(request!.id, { type, fileName: up.fileName, dataUri: up.dataUri })
   308→          savedRequest = r.request
   309→        }
   310→        const v = await api.verifyDocuments(request!.id)
   311→        savedRequest = v.request
   312→      }
   313→
   314→      toast({
   315→        title: 'Водитель сохранён в справочнике',
   316→        description: attachToRequest
   317→          ? `${request!.number}: карточка заполнена нейросетью, документы прикреплены и отправлены на проверку`
   318→          : 'Карточка заполнена нейросетью по загруженным документам',
   319→      })
   320→      onSaved({ driverId, truckId, driverName: draftDriver.fullName, request: savedRequest })
   321→      onOpenChange(false)
   322→    } catch (e) {
   323→      toast({ title: 'Не удалось сохранить водителя', description: e instanceof Error ? e.message : 'Неизвестная ошибка', variant: 'destructive' })
   324→    } finally {
   325→      setSaving(false)
   326→    }
   327→  }
   328→
   329→  const badDocs: IntakeDocVerdict[] = result?.documents.filter((d) => d.status === 'REUPLOAD' || d.status === 'REJECTED') ?? []
   330→  const canSave = !!(draftDriver?.fullName && draftDriver?.license && carrierId) && !saving
   331→
   332→  return (
   333→    <Dialog open={open} onOpenChange={onOpenChange}>
   334→      <DialogContent className="max-h-[88vh] overflow-y-auto sm:max-w-2xl">
   335→        <DialogHeader>
   336→          <DialogTitle className="flex items-center gap-2">
   337→            <ScanLine className="h-5 w-5 text-blue-700" aria-hidden />
   338→            Новый водитель — по документам, с нейросетью
   339→          </DialogTitle>
   340→          <DialogDescription>
   341→            {request
   342→              ? `Заявка ${request.number}: загрузите документы — нейросеть прочитает их, сама заполнит карточку водителя и тягача и сверит данные. Останется подтвердить.`
   343→              : 'Загрузите документы — нейросеть прочитает паспорт, ВУ и СТС, сама заполнит карточку водителя и тягача и проверит данные на противоречия.'}
   344→          </DialogDescription>
   345→        </DialogHeader>
   346→
   347→        {/* индикатор шагов */}
   348→        <ol className="flex flex-wrap items-center gap-1.5 text-xs font-medium" aria-label="Шаги добавления водителя">
   349→          {[
   350→            { n: 1, label: 'Документы', active: step === 'docs' },
   351→            { n: 2, label: 'Нейросеть', active: step === 'scan' },
   352→            { n: 3, label: 'Подтверждение', active: step === 'review' },
   353→          ].map((s, i) => (
   354→            <li key={s.n} className="flex items-center gap-1.5">
   355→              {i > 0 && <ArrowRight className="h-3 w-3 text-slate-300" aria-hidden />}
   356→              <span className={`flex items-center gap-1.5 rounded-full border px-2.5 py-1 ${s.active ? 'border-blue-200 bg-blue-50 text-blue-800' : 'border-slate-200 bg-slate-50 text-slate-500'}`}>
   357→                <span className={`flex h-4 w-4 items-center justify-center rounded-full text-[10px] ${s.active ? 'bg-blue-700 text-white' : 'bg-slate-300 text-white'}`} aria-hidden>{s.n}</span>
   358→                {s.label}
   359→              </span>
   360→            </li>
   361→          ))}
   362→        </ol>
   363→
   364→        {/* ——— Шаг 1: документы ——— */}
   365→        {step === 'docs' && (
   366→          <div className="space-y-4">
   367→            {!lockCarrier && (
   368→              <div className="space-y-1.5">
   369→                <Label htmlFor="intake-carrier">Перевозчик</Label>
   370→                <Select value={carrierId} onValueChange={setCarrierId}>
   371→                  <SelectTrigger id="intake-carrier" aria-label="Выбор перевозчика">
   372→                    <SelectValue placeholder="Выберите перевозчика" />
   373→                  </SelectTrigger>
   374→                  <SelectContent>
   375→                    {carriers.map((c) => (
   376→                      <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>
   377→                    ))}
   378→                  </SelectContent>
   379→                </Select>
   380→              </div>
   381→            )}
   382→
   383→            <div className="grid grid-cols-1 gap-3 sm:grid-cols-2">
   384→              {([
   385→                { type: 'PASSPORT_PHOTO' as DocType, required: true, icon: IdCard, hint: 'Страница с фото' },
   386→                { type: 'LICENSE' as DocType, required: true, icon: Car, hint: 'Лицевая сторона' },
   387→                { type: 'STS' as DocType, required: false, icon: Truck, hint: 'Необязательно — добавим и тягач' },
   388→                { type: 'PASSPORT_REG' as DocType, required: false, icon: MapPin, hint: 'Необязательно — для комплекта заявки' },
   389→              ]).map(({ type, required, icon: Icon, hint }) => {
   390→                const up = uploads[type]
   391→                const key = `intake:${type}`
   392→                return (
   393→                  <div key={type} className={`flex flex-col rounded-xl border p-3 ${up ? 'border-slate-200 bg-white' : 'border-dashed border-slate-300 bg-slate-50/60'}`}>
   394→                    <div className="mb-2 flex items-start justify-between gap-2">
   395→                      <span className="flex items-center gap-2">
   396→                        <span className="flex h-8 w-8 shrink-0 items-center justify-center rounded-lg bg-blue-50 text-blue-700" aria-hidden>
   397→                          <Icon className="h-4 w-4" />
   398→                        </span>
   399→                        <span className="text-xs font-semibold leading-tight text-slate-800">
   400→                          {DOC_TYPE_LABEL[type]}
   401→                          {!required && <span className="block text-[10px] font-normal text-slate-500">{hint}</span>}
   402→                        </span>
   403→                      </span>
   404→                      {required && <Badge variant="outline" className="border-blue-200 bg-blue-50 text-[10px] text-blue-700">Обязательно</Badge>}
   405→                    </div>
   406→                    {up && (
   407→                      <img src={up.dataUri} alt={`Загруженный документ «${DOC_TYPE_LABEL[type]}»`} className="mb-2 h-20 w-full rounded-md border border-slate-200 object-cover" />
   408→                    )}
   409→                    <input
   410→                      ref={(el) => { fileInputs.current[type] = el }}
   411→                      type="file"
   412→                      accept="image/*"
   413→                      className="hidden"
   414→                      aria-label={`Загрузить ${DOC_TYPE_LABEL[type]}`}
   415→                      onChange={(e) => {
   416→                        const f = e.target.files?.[0]
   417→                        e.target.value = ''
   418→                        if (f) void handleFile(type, f)
   419→                      }}
   420→                    />
   421→                    <Button
