   724→   724→function distM(a: LatLng, b: LatLng): number {
   725→   725→  const R = 6371000
   726→   726→  const dLat = ((b.lat - a.lat) * Math.PI) / 180
   727→   727→  const dLon = ((b.lng - a.lng) * Math.PI) / 180
   728→   728→  const la1 = (a.lat * Math.PI) / 180
   729→   729→  const la2 = (b.lat * Math.PI) / 180
   730→   730→  const h = Math.sin(dLat / 2) ** 2 + Math.cos(la1) * Math.cos(la2) * Math.sin(dLon / 2) ** 2
   731→   731→  return Math.round(2 * R * Math.asin(Math.sqrt(h)))
   732→   732→}
